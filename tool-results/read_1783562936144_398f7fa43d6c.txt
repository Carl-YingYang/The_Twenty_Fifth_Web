    99→/** Count reservations visible under a given tab (History aggregates 4 statuses). */
   100→function countForTab(tab: string, reservations: Reservation[]): number {
   101→  if (tab === "ALL") return reservations.length;
   102→  if (tab === "HISTORY") {
   103→    return reservations.filter((r) =>
   104→      HISTORY_STATUSES.includes(r.status as BookingStatus)
   105→    ).length;
   106→  }
   107→  return reservations.filter((r) => r.status === tab).length;
   108→}
   109→
   110→/** Build a compact page range with ellipses for large page counts. */
   111→function getPageRange(
   112→  current: number,
   113→  total: number
   114→): (number | "ellipsis")[] {
   115→  if (total <= 7) {
   116→    return Array.from({ length: total }, (_, i) => i + 1);
   117→  }
   118→  if (current <= 4) {
   119→    return [1, 2, 3, 4, 5, "ellipsis", total];
   120→  }
   121→  if (current >= total - 3) {
   122→    return [1, "ellipsis", total - 4, total - 3, total - 2, total - 1, total];
   123→  }
   124→  return [1, "ellipsis", current - 1, current, current + 1, "ellipsis", total];
   125→}

   585→function RowActions({
   586→  reservation,

   588→  onMutate,
