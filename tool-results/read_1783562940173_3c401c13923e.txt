    98→    98→export function RoomsAdmin() {
    99→    99→  const qc = useQueryClient();
   100→   100→  const { data, isLoading } = useQuery({
   101→   101→    queryKey: ["admin-rooms"],
   102→   102→    queryFn: () => apiFetch<{ rooms: Room[] }>("/api/rooms"),
   103→   103→  });
   104→   104→  const rooms = (data?.rooms ?? []).filter((r) => r.isActive);
   105→   105→
   106→   106→  const [editing, setEditing] = useState<Room | null>(null);
   107→   107→  const [creating, setCreating] = useState(false);
   108→   108→  const [deleting, setDeleting] = useState<Room | null>(null);
   109→   109→  const [page, setPage] = useState(1);
   110→   110→
   111→   111→  // Client-side pagination
   112→   112→  const totalPages = Math.max(1, Math.ceil(rooms.length / ROOMS_PAGE_SIZE));
   113→   113→  const currentPage = Math.min(page, totalPages);
   114→   114→  const startIdx = (currentPage - 1) * ROOMS_PAGE_SIZE;
   115→   115→  const endIdx = Math.min(startIdx + ROOMS_PAGE_SIZE, rooms.length);
   116→   116→  const pagedRooms = rooms.slice(startIdx, endIdx);
   117→   117→
   118→   118→  const goToPage = (p: number) => {
   119→   119→    setPage(Math.max(1, Math.min(p, totalPages)));
   120→   120→  };
   121→   121→
   122→   122→  const statusMutation = useMutation({
   123→   123→    mutationFn: ({ id, status }: { id: string; status: RoomStatus }) =>
   124→   124→      apiFetch<{ room: Room }>(`/api/rooms/${id}`, {
   125→   125→        method: "PATCH",
   126→   126→        body: JSON.stringify({ status }),
   127→   127→      }),
   128→   128→    onSuccess: (data) => {
   129→   129→      qc.invalidateQueries({ queryKey: ["admin-rooms"] });
   130→   130→      qc.invalidateQueries({ queryKey: ["admin-calendar"] });
   131→   131→      qc.invalidateQueries({ queryKey: ["admin-dashboard"] });
   132→   132→      toast.success(`${data.room.name} marked as ${ROOM_STATUS_CONFIG[data.room.status].label}.`);
   133→   133→    },
   134→   134→    onError: (err) => {
   135→   135→      const message =
   136→   136→        err instanceof ApiError ? err.message : "Couldn't update room.";
   137→   137→      toast.error(message);
   138→   138→    },
   139→   139→  });
   140→   140→
   141→   141→  const deleteMutation = useMutation({
   142→   142→    mutationFn: (id: string) =>
   143→   143→      apiFetch(`/api/rooms/${id}`, { method: "DELETE" }),
   144→   144→    onSuccess: () => {
   145→   145→      qc.invalidateQueries({ queryKey: ["admin-rooms"] });
   146→   146→      toast.success("Room removed.");
   147→   147→      setDeleting(null);
   148→   148→    },
   149→   149→    onError: (err) => {
   150→   150→      const message =
   151→   151→        err instanceof ApiError ? err.message : "Couldn't remove room.";
   152→   152→      toast.error(message);
   153→   153→    },
   154→   154→  });
   155→   155→
   156→   156→  return (
   157→   157→    <AdminLayout title="The Villa" subtitle="Manage rooms, photos, and availability">
   158→   158→      <div className="mb-5 flex items-center justify-between gap-3 pb-6">
   159→   159→        <div className="min-w-0">
   160→   160→          <div className="eyebrow">{rooms.length} spaces</div>
   161→   161→          <h2 className="mt-1 truncate font-display text-2xl font-medium tracking-tight">
   162→   162→            The Villa at {`The Twenty-Fifth`}
   163→   163→          </h2>
   164→   164→        </div>
   165→   165→        <Button
   166→   166→          onClick={() => setCreating(true)}
   167→   167→          className="shrink-0 bg-primary text-primary-foreground hover:bg-primary/90"
   168→   168→        >
   169→   169→          <Plus className="size-4" />
   170→   170→          <span className="hidden sm:inline">Add Room</span>
   171→   171→        </Button>
   172→   172→      </div>
   173→   173→
   174→   174→      {isLoading ? (
   175→   175→        <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-3">
   176→   176→          {Array.from({ length: 3 }).map((_, i) => (
   177→   177→            <Skeleton key={i} className="h-72 w-full rounded-lg" />
   178→   178→          ))}
   179→   179→        </div>
   180→   180→      ) : rooms.length === 0 ? (
   181→   181→        <EmptyState
   182→   182→          icon={Home}
   183→   183→          title="No rooms yet"
   184→   184→          description="Add your first room to start taking reservations."
   185→   185→          action={
   186→   186→            <Button
   187→   187→              onClick={() => setCreating(true)}
   188→   188→              className="bg-primary text-primary-foreground hover:bg-primary/90"
   189→   189→            >
   190→   190→              <Plus className="size-4" />
   191→   191→              Add Room
   192→   192→            </Button>
   193→   193→          }
   194→   194→        />
   195→   195→      ) : (
   196→   196→        <>
   197→   197→          <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-3 pb-6">
   198→   198→            {pagedRooms.map((room) => (
   199→   199→              <RoomCard
   200→   200→                key={room.id}
   201→   201→                room={room}
   202→   202→                onEdit={() => setEditing(room)}
   203→   203→                onDelete={() => setDeleting(room)}
   204→   204→                onStatusChange={(s) =>
   205→   205→                  statusMutation.mutate({ id: room.id, status: s })
   206→   206→                }
   207→   207→                pending={statusMutation.isPending}
   208→   208→              />
   209→   209→            ))}
   210→   210→          </div>
   211→   211→
   212→   212→          {/* Pagination footer */}
   213→   213→          {totalPages > 1 && (
   214→   214→            <div className="flex flex-col items-center gap-3 border-t border-border pt-4 sm:flex-row sm:justify-between">
   215→   215→              <p className="text-xs text-muted-foreground">
   216→   216→                Showing {startIdx + 1}–{endIdx} of {rooms.length} rooms
   217→   217→              </p>
   218→   218→              <Pagination className="sm:justify-end">
   219→   219→                <PaginationContent>
   220→   220→                  <PaginationItem>
   221→   221→                    <button
   222→   222→                      type="button"
   223→   223→                      onClick={() => goToPage(1)}
   224→   224→                      disabled={currentPage === 1}
   225→   225→                      className="inline-flex size-8 items-center justify-center rounded-md text-muted-foreground transition-colors hover:bg-muted hover:text-foreground disabled:pointer-events-none disabled:opacity-40"
   226→   226→                      aria-label="First page"
   227→   227→                    >
   228→   228→                      <ChevronsLeft className="size-4" />
   229→   229→                    </button>
   230→   230→                  </PaginationItem>
   231→   231→                  <PaginationItem>
   232→   232→                    <PaginationPrevious
   233→   233→                      onClick={() => goToPage(currentPage - 1)}
   234→   234→                      aria-disabled={currentPage === 1}
   235→   235→                      className={currentPage === 1 ? "pointer-events-none opacity-40" : ""}
   236→   236→                    />
   237→   237→                  </PaginationItem>

   240→   240→                    // Show first, last, current, and neighbors; ellipsis for gaps
   241→   241→                    if (
   242→   242→                      p === 1 ||
   243→   243→                      p === totalPages ||
   244→   244→                      Math.abs(p - currentPage) <= 1
   245→   245→                    ) {
   246→   246→                      return (
   247→   247→                        <PaginationItem key={p}>
   248→   248→                          <PaginationLink
   249→   249→                            isActive={p === currentPage}

   253→   253→                          </PaginationLink>
   254→   254→                        </PaginationItem>
   255→   255→                      );

   257→   257→                    if (p === currentPage - 2 || p === currentPage + 2) {
   258→   258→                      return (
   259→   259→                        <PaginationItem key={p} className="text-muted-foreground">
   260→   260→                          <span className="px-1">…</span>

   262→   262→                      );
   263→   263→                    }
   264→   264→                    return null;
   265→   265→                  })}
   266→   266→                  <PaginationItem>
   267→   267→                    <PaginationNext
   268→   268→                      onClick={() => goToPage(currentPage + 1)}
   269→   269→                      aria-disabled={currentPage === totalPages}
   270→   270→                      className={currentPage === totalPages ? "pointer-events-none opacity-40" : ""}
   271→   271→                    />
   272→   272→                  </PaginationItem>
   273→   273→                  <PaginationItem>
   274→   274→                    <button
   275→   275→                      type="button"
   276→   276→                      onClick={() => goToPage(totalPages)}
   277→   277→                      disabled={currentPage === totalPages}
   278→   278→                      className="inline-flex size-8 items-center justify-center rounded-md text-muted-foreground transition-colors hover:bg-muted hover:text-foreground disabled:pointer-events-none disabled:opacity-40"
   279→   279→                      aria-label="Last page"
   280→   280→                    >
   281→   281→                      <ChevronsRight className="size-4" />
   282→   282→                    </button>
   283→   283→                  </PaginationItem>
   284→   284→                </PaginationContent>
   285→   285→              </Pagination>
   286→   286→            </div>
   287→   287→          )}
   288→   288→        </>
   289→   289→      )}
   290→   290→
   291→   291→      {/* Edit dialog */}
   292→   292→      {editing && (
   293→   293→        <RoomFormDialog
   294→   294→          room={editing}
   295→   295→          open={!!editing}
   296→   296→          onClose={() => setEditing(null)}
   297→   297→          onSaved={() => {
   298→   298→            setEditing(null);
   299→   299→            qc.invalidateQueries({ queryKey: ["admin-rooms"] });
   300→   300→          }}
   301→   301→        />
   302→   302→      )}
   303→   303→
   304→   304→      {/* Create dialog */}
   305→   305→      <RoomFormDialog
   306→   306→        open={creating}
   307→   307→        onClose={() => setCreating(false)}
   308→   308→        onSaved={() => {
   309→   309→          setCreating(false);
   310→   310→          qc.invalidateQueries({ queryKey: ["admin-rooms"] });
   311→   311→        }}
   312→   312→      />
   313→   313→
   314→   314→      {/* Delete confirm */}
   315→   315→      <AlertDialog
   316→   316→        open={!!deleting}
   317→   317→        onOpenChange={(o) => !o && setDeleting(null)}
   318→   318→      >
   319→   319→        <AlertDialogContent>
   320→   320→          <AlertDialogHeader>
   321→   321→            <AlertDialogTitle>Remove this room?</AlertDialogTitle>
   322→   322→            <AlertDialogDescription>
   323→   323→              {deleting?.name} will be hidden from the website and calendar.
   324→   324→              Existing reservations are kept. This cannot be undone.
   325→   325→            </AlertDialogDescription>
   326→   326→          </AlertDialogHeader>
   327→   327→          <AlertDialogFooter>
   328→   328→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   329→   329→            <AlertDialogAction
   330→   330→              className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
   331→   331→              onClick={() => deleting && deleteMutation.mutate(deleting.id)}
   332→   332→              disabled={deleteMutation.isPending}
   333→   333→            >
   334→   334→              {deleteMutation.isPending ? "Removing…" : "Remove room"}
   335→   335→            </AlertDialogAction>
   336→   336→          </AlertDialogFooter>
   337→   337→        </AlertDialogContent>
   338→   338→      </AlertDialog>
   339→   339→    </AdminLayout>
   340→   340→  );
   341→   341→}

   344→   344→  room,
   345→   345→  onEdit,
   346→   346→  onDelete,
   347→   347→  onStatusChange,
   348→   348→  pending,

   351→   351→  onEdit: () => void;

   353→   353→  onStatusChange: (s: RoomStatus) => void;
   354→   354→  pending: boolean;
   355→   355→}) {
   356→   356→  const image = room.images?.[0]?.url;
   357→   357→  const cfg = ROOM_STATUS_CONFIG[room.status as string] ?? ROOM_STATUS_CONFIG.AVAILABLE;
   358→   358→
   359→   359→  return (
   360→   360→    <Card className="overflow-hidden rounded-lg border border-border shadow-card">
   361→   361→      <div className="relative aspect-[4/3] w-full overflow-hidden bg-muted">

   366→   366→            className="size-full object-cover"

   500→   500→function parseImageAlt(altText: string): { category: ImageCategory; caption: string } {
   501→   501→  if (!altText) return { category: "Other", caption: "" };
   502→   502→  const sepIdx = altText.indexOf("::");
   503→   503→  if (sepIdx === -1) return { category: "Other", caption: altText };
   504→   504→  const cat = altText.slice(0, sepIdx) as ImageCategory;
   505→   505→  const cap = altText.slice(sepIdx + 2);
   506→   506→  const valid = IMAGE_CATEGORIES.some((c) => c.value === cat);
   507→   507→  return { category: valid ? cat : "Other", caption: cap };
   508→   508→}
   509→   509→
   510→   510→function formatImageAlt(category: ImageCategory, caption: string): string {
   511→   511→  return `${category}::${caption.trim()}`;
   512→   512→}
   513→   513→
   514→   514→function getCategoryConfig(value: ImageCategory) {
   515→   515→  return IMAGE_CATEGORIES.find((c) => c.value === value) ?? IMAGE_CATEGORIES[IMAGE_CATEGORIES.length - 1];
   516→   516→}
