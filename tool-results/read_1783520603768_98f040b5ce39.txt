    87→    87→
    88→    88→  const {
    89→    89→    data: roomsData,
    90→    90→    isLoading: roomsLoading,
    91→    91→  } = useQuery({
    92→    92→    queryKey: ["rooms", "booking-flow"],
    93→    93→    queryFn: () => apiFetch<RoomsResponse>("/api/rooms"),
    94→    94→  });
    95→    95→  const rooms = roomsData?.rooms ?? [];
    96→    96→  const selectedRoom = rooms.find((r) => r.id === booking.selectedRoomId);
    97→    97→
    98→    98→  const createMutation = useMutation({

   100→   100→      roomId: string;
   101→   101→      checkIn: string;
   102→   102→      checkOut: string;
   103→   103→      adults: number;
   104→   104→      children: number;
   105→   105→      guest: GuestInfoInput;
   106→   106→      specialRequests?: string;
   107→   107→    }) =>

   109→   109→        method: "POST",
   110→   110→        body: JSON.stringify(payload),
   111→   111→      }),
   112→   112→    onSuccess: (data) => {
   113→   113→      setConfirmed(data.reservation);
   114→   114→      setStep(3);
   115→   115→      toast.success("Booking request received!");

   117→   117→    onError: (err) => {
   118→   118→      const msg =
   119→   119→        err instanceof ApiError
   120→   120→          ? err.message
   121→   121→          : "Something went wrong. Please try again.";
   122→   122→      toast.error(msg);
   123→   123→    },
   124→   124→  });

   141→   141→        reservation={confirmed}
   142→   142→        onReset={() => {

   144→   144→          setConfirmed(null);
   145→   145→          setStep(1);
   146→   146→          navigate("home");
   147→   147→        }}
   148→   148→      />
   149→   149→    );
   150→   150→  }
   151→   151→
   152→   152→  return (
   153→   153→    <div className="pt-8 sm:pt-12">

   157→   157→          <div className="flex items-center justify-center gap-2 sm:gap-4">
   158→   158→            {STEPS.map((s, i) => {
   159→   159→              const active = step === s.value;
   160→   160→              const done = step > s.value;
   161→   161→              return (
   162→   162→                <React.Fragment key={s.value}>
   163→   163→                  <button
   164→   164→                    type="button"
   165→   165→                    onClick={() => {
   166→   166→                      // Allow going back to previous steps.
   167→   167→                      if (s.value < step) setStep(s.value);
   168→   168→                    }}

   175→   175→                        active

   186→   186→                        "text-sm font-medium transition-colors",
   187→   187→                        active
   188→   188→                          ? "text-foreground"
   189→   189→                          : "text-muted-foreground"
   190→   190→                      )}
   191→   191→                    >
   192→   192→                      {s.label}
   193→   193→                    </span>
   194→   194→                  </button>
   195→   195→                  {i < STEPS.length - 1 && (
   196→   196→                    <span
   197→   197→                      className={cn(
   198→   198→                        "h-px w-8 sm:w-16",
   199→   199→                        step > s.value ? "bg-primary/40" : "bg-border"
   200→   200→                      )}
   201→   201→                    />
   202→   202→                  )}
   203→   203→                </React.Fragment>
   204→   204→              );

   206→   206→          </div>
   207→   207→        </div>
   208→   208→      </div>
   209→   209→
   210→   210→      <div className="py-8 sm:py-12">
   211→   211→        <div className="container-luxury">
   212→   212→          {/* ============================================================
   213→   213→              STEP 1 — Dates & Guests + Choose your stay
   214→   214→          ============================================================ */}

   216→   216→            <Step1Dates
   217→   217→              booking={booking}
   218→   218→              setSearch={setSearch}
   219→   219→              selectRoom={selectRoom}
   220→   220→              rooms={rooms}
   221→   221→              roomsLoading={roomsLoading}
   222→   222→              selectedRoom={selectedRoom}
   223→   223→              onContinue={() => setStep(2)}
   224→   224→            />
   225→   225→          )}
   226→   226→
   227→   227→          {/* ============================================================
   228→   228→              STEP 2 — Your Details
   229→   229→          ============================================================ */}
   230→   230→          {step === 2 && (
   231→   231→            <Step2Details
   232→   232→              defaultValues={guestInfo}
   233→   233→              onBack={() => setStep(1)}
   234→   234→              onContinue={(values) => {
   235→   235→                setGuestInfo(values);
   236→   236→                setStep(3);
   237→   237→              }}
   238→   238→            />

   240→   240→

   248→   248→              selectedRoom={selectedRoom}
   249→   249→              onBack={() => setStep(2)}
   250→   250→              onConfirm={() => {

   252→   252→                  toast.error("Please choose your stay first");

   254→   254→                  return;

   257→   257→                  roomId: selectedRoom.id,
   258→   258→                  checkIn: booking.checkIn,
   259→   259→                  checkOut: booking.checkOut,
   260→   260→                  adults: booking.adults,
   261→   261→                  children: booking.children,
   262→   262→                  guest: guestInfo,
   263→   263→                  specialRequests: guestInfo.specialRequests || undefined,
   264→   264→                });
   265→   265→              }}
   266→   266→              submitting={createMutation.isPending}
   267→   267→            />
   268→   268→          )}
   269→   269→        </div>
   270→   270→      </div>
   271→   271→    </div>
