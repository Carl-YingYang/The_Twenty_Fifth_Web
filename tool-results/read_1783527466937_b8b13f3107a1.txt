   410→   410→function RowActions({
   411→   411→  reservation,
   412→   412→  onView,
   413→   413→  onMutate,
   414→   414→  pending,
   415→   415→  fullWidth,
   416→   416→}: {
   417→   417→  reservation: Reservation;
   418→   418→  onView: () => void;
   419→   419→  onMutate: (status: BookingStatus) => void;
   420→   420→  pending: boolean;
   421→   421→  fullWidth?: boolean;
   422→   422→}) {
   423→   423→  const s = reservation.status as BookingStatus;
   424→   424→  return (
   425→   425→    <div
   426→   426→      className={cn(
   427→   427→        "flex items-center justify-end gap-1.5",
   428→   428→        fullWidth && "w-full"
   429→   429→      )}
   430→   430→    >
   431→   431→      {s === "PENDING" && (
   432→   432→        <>
   433→   433→          <Button
   434→   434→            size="sm"
   435→   435→            className="h-8 bg-emerald-600 text-white hover:bg-emerald-700"
   436→   436→            disabled={pending}
   437→   437→            onClick={() => onMutate("CONFIRMED")}
   438→   438→          >
   439→   439→            <Check className="size-3.5" />
   440→   440→            Approve
   441→   441→          </Button>
   442→   442→          <Button
   443→   443→            size="sm"
   444→   444→            variant="outline"
   445→   445→            className="h-8 border-red-300 text-red-700 hover:bg-red-50"
   446→   446→            disabled={pending}
   447→   447→            onClick={() => onMutate("REJECTED")}
   448→   448→          >
   449→   449→            <X className="size-3.5" />
   450→   450→            Decline
   451→   451→          </Button>
   452→   452→        </>
   453→   453→      )}
   454→   454→      {s === "CONFIRMED" && (
   455→   455→        <Button
   456→   456→          size="sm"
   457→   457→          className="h-8 bg-primary text-primary-foreground hover:bg-primary/90"

   461→   461→          <LogIn className="size-3.5" />
   462→   462→          Check in

   464→   464→      )}

   468→   468→          className="h-8 bg-primary text-primary-foreground hover:bg-primary/90"
   469→   469→          disabled={pending}
   470→   470→          onClick={() => onMutate("COMPLETED")}
   471→   471→        >
   472→   472→          <LogOut className="size-3.5" />
   473→   473→          Check out
   474→   474→        </Button>
   475→   475→      )}
   476→   476→      {(s === "COMPLETED" ||
   477→   477→        s === "CANCELLED" ||
   478→   478→        s === "REJECTED" ||
   479→   479→        s === "NO_SHOW") && (
   480→   480→        <Button
   481→   481→          size="sm"
