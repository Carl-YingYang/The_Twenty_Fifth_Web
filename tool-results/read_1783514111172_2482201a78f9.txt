    55→interface AvailabilityResponse {
    56→  available: AvailableRoom[];
    57→  unavailable: AvailableRoom[];
    58→  nights: number;
    59→}
    60→interface ReservationResponse {
    61→  reservation: Reservation;
    62→}
    63→
    64→const STEPS = [
    65→  { id: 1, label: "Dates", short: "Search" },
    66→  { id: 2, label: "Room", short: "Select" },
    67→  { id: 3, label: "Guest", short: "Details" },
    68→  { id: 4, label: "Review", short: "Confirm" },
    69→  { id: 5, label: "Done", short: "Confirmed" },
    70→];
    71→
    72→function defaultDate(daysFromNow: number): string {
    73→  const d = new Date();
    74→  d.setDate(d.getDate() + daysFromNow);
    75→  return d.toISOString().slice(0, 10);
    76→}
    77→
    78→export function BookingFlow() {
    79→  const navigate = useViewStore((s) => s.navigate);
    80→  const params = useViewStore((s) => s.params);
    81→  const booking = useBookingStore();
    82→  const setSearch = useBookingStore((s) => s.setSearch);
    83→  const selectRoom = useBookingStore((s) => s.selectRoom);
    84→  const resetBooking = useBookingStore((s) => s.reset);
    85→
    86→  const [step, setStep] = React.useState(1);
    87→  const [checkIn, setCheckIn] = React.useState(booking.checkIn || defaultDate(1));
    88→  const [checkOut, setCheckOut] = React.useState(booking.checkOut || defaultDate(3));
    89→  const [adults, setAdults] = React.useState(String(booking.adults || 2));
    90→  const [children, setChildren] = React.useState(String(booking.children || 0));
    91→  const [selectedRoomId, setSelectedRoomId] = React.useState<string | null>(
    92→    booking.selectedRoomId || params.roomId || null
    93→  );
    94→  const [guestInfo, setGuestInfo] = React.useState<GuestInfoInput | null>(null);
    95→  const [confirmedReservation, setConfirmedReservation] = React.useState<Reservation | null>(null);
    96→
    97→  const nights = nightsBetween(checkIn, checkOut);
    98→
    99→  // Availability query (only triggered on step 2)
   100→  const availabilityQuery = useQuery({
   101→    queryKey: ["availability", "booking", checkIn, checkOut, adults, children],
   102→    queryFn: () =>
   103→      apiFetch<AvailabilityResponse>(
   104→        `/api/rooms/availability?checkIn=${checkIn}&checkOut=${checkOut}&adults=${adults}&children=${children}`
   105→      ),
   106→    enabled: step === 2 && !!checkIn && !!checkOut && new Date(checkOut) > new Date(checkIn),
   107→  });
   108→
   109→  const selectedRoom =
   110→    availabilityQuery.data?.available.find((r) => r.id === selectedRoomId) ??
   111→    availabilityQuery.data?.unavailable.find((r) => r.id === selectedRoomId) ??
   112→    null;
   113→
   114→  // Create reservation mutation
   115→  const createMutation = useMutation({
   116→    mutationFn: (payload: {
   117→      roomId: string;
   118→      checkIn: string;
   119→      checkOut: string;
   120→      adults: number;
   121→      children: number;
   122→      guest: GuestInfoInput;
   123→      specialRequests?: string;
   124→    }) =>
   125→      apiFetch<ReservationResponse>("/api/reservations", {
   126→        method: "POST",
   127→        body: JSON.stringify(payload),
   128→      }),
   129→    onSuccess: (data) => {
   130→      setConfirmedReservation(data.reservation);
   131→      setStep(5);
   132→      toast.success("Reservation created!", {
   133→        description: `Reference ${data.reservation.referenceNo}`,
   134→      });
   135→    },
   136→    onError: (err) => {
   137→      const message = err instanceof ApiError ? err.message : "Failed to create reservation";
   138→      toast.error(message);
   139→      setStep(4);
   140→    },
   141→  });
   142→
   143→  // ---- Step handlers ----
   144→
   145→  const onSearchAvailability = () => {
   146→    if (!checkIn || !checkOut) {
   147→      toast.error("Please select your check-in and check-out dates");
   148→      return;
   149→    }
   150→    if (new Date(checkOut) <= new Date(checkIn)) {
   151→      toast.error("Check-out must be after check-in");
   152→      return;
   153→    }
   154→    setSearch({
   155→      checkIn,
   156→      checkOut,
   157→      adults: parseInt(adults, 10),
   158→      children: parseInt(children, 10),
   159→    });
   160→    setStep(2);
   161→  };
   162→
   163→  const onSelectRoom = (room: AvailableRoom) => {
   164→    if (!room.isAvailable) {
   165→      toast.error("This room is not available for the selected dates");
   166→      return;
   167→    }
   168→    setSelectedRoomId(room.id);
   169→    selectRoom(room.id);
   170→    setStep(3);
   171→  };
   172→
   173→  const onGuestInfoSubmit = (values: GuestInfoInput) => {
   174→    setGuestInfo(values);
   175→    setStep(4);
   176→  };
   177→
   178→  const onConfirmReservation = () => {
   179→    if (!selectedRoomId || !guestInfo) {
   180→      toast.error("Missing reservation details");
   181→      return;
   182→    }
   183→    setStep(5); // show loading state
   184→    createMutation.mutate({
   185→      roomId: selectedRoomId,
   186→      checkIn,
   187→      checkOut,
   188→      adults: parseInt(adults, 10),
   189→      children: parseInt(children, 10),
   190→      guest: guestInfo,
   191→      specialRequests: guestInfo.specialRequests,
   192→    });
   193→  };
   194→
   195→  const restart = () => {
   196→    resetBooking();
   197→    setStep(1);
   198→    setCheckIn(defaultDate(1));
   199→    setCheckOut(defaultDate(3));
   200→    setAdults("2");
   201→    setChildren("0");
   202→    setSelectedRoomId(null);
   203→    setGuestInfo(null);
   204→    setConfirmedReservation(null);
   205→    navigate("home");
   206→  };
   207→
   208→  return (
   209→    <div className="pt-16 md:pt-20">
   210→      {/* Header */}
   211→      <section className="border-b border-border/60 bg-section">
   212→        <div className="container-luxury py-8">
   213→          <div className="flex items-center justify-between gap-4">
   214→            <div>
   215→              <p className="text-xs font-medium uppercase tracking-[0.25em] text-primary">
   216→                Reserve Your Stay
   217→              </p>
   218→              <h1 className="mt-1 font-display text-3xl font-semibold tracking-tight sm:text-4xl">
   219→                Book Verdara
   220→              </h1>
   221→            </div>
   222→            {step > 1 && step < 5 && (
   223→              <Button
   224→                variant="ghost"
   225→                onClick={() => setStep(step - 1)}
   226→                className="rounded-full"
   227→              >
   228→                <ArrowLeft className="h-4 w-4" />
   229→                Back
   230→              </Button>
   231→            )}
   232→          </div>
   233→        </div>
   234→      </section>
   235→
   236→      {/* Progress indicator */}
   237→      {step < 5 && (
   238→        <div className="border-b border-border/60 bg-background">
   239→          <div className="container-luxury py-5">
   240→            <ol className="flex items-center justify-between gap-1 overflow-x-auto">
   241→              {STEPS.slice(0, 4).map((s, idx) => {
   242→                const status =
   243→                  step === s.id ? "current" : step > s.id ? "complete" : "upcoming";
   244→                return (
   245→                  <li key={s.id} className="flex flex-1 items-center gap-3">
   246→                    <button
   247→                      onClick={() => {
   248→                        if (s.id < step) setStep(s.id);
   249→                      }}
   250→                      disabled={s.id >= step}
   251→                      className="flex items-center gap-3"
   252→                    >
   253→                      <span
   254→                        className={cn(
   255→                          "flex h-9 w-9 shrink-0 items-center justify-center rounded-full border-2 text-sm font-medium transition-all",
   256→                          status === "current" && "border-primary bg-primary text-white",
   257→                          status === "complete" && "border-primary bg-primary/10 text-primary",
   258→                          status === "upcoming" && "border-border bg-card text-muted-foreground"
   259→                        )}
   260→                      >
   261→                        {status === "complete" ? <Check className="h-4 w-4" /> : idx + 1}
   262→                      </span>
   263→                      <div className="hidden sm:block">
   264→                        <div
   265→                          className={cn(
   266→                            "text-xs uppercase tracking-wider",
   267→                            status === "upcoming" ? "text-muted-foreground" : "text-foreground"
   268→                          )}
   269→                        >
   270→                          Step {idx + 1}
   271→                        </div>
   272→                        <div
   273→                          className={cn(
   274→                            "text-sm font-medium",
   275→                            status === "current" && "text-primary",
   276→                            status === "complete" && "text-foreground",
   277→                            status === "upcoming" && "text-muted-foreground"
   278→                          )}
   279→                        >
   280→                          {s.label}
   281→                        </div>
   282→                      </div>
   283→                    </button>
   284→                    {idx < 3 && (
   285→                      <div
   286→                        className={cn(
   287→                          "ml-1 h-px flex-1",
   288→                          step > s.id ? "bg-primary" : "bg-border"
   289→                        )}
   290→                      />
   291→                    )}
   292→                  </li>
   293→                );
   294→              })}
   295→            </ol>
   296→          </div>
   297→        </div>
   298→      )}
   299→
   300→      <section className="py-10 sm:py-14">
   301→        <div className="container-luxury">
   302→          <AnimatePresence mode="wait">
   303→            {/* STEP 1: Search */}
   304→            {step === 1 && (
   305→              <motion.div
   306→                key="step1"
   307→                initial={{ opacity: 0, y: 10 }}
   308→                animate={{ opacity: 1, y: 0 }}
   309→                exit={{ opacity: 0, y: -10 }}
   310→                transition={{ duration: 0.3 }}
   311→                className="mx-auto max-w-3xl"
   312→              >
   313→                <Card className="rounded-2xl border-border/60 p-6 shadow-luxury sm:p-10">
   314→                  <div className="text-center">
   315→                    <span className="mx-auto flex h-12 w-12 items-center justify-center rounded-full bg-primary/10 text-primary">
   316→                      <CalendarIcon className="h-5 w-5" />
   317→                    </span>
   318→                    <h2 className="mt-4 font-display text-2xl font-semibold tracking-tight">
   319→                      When would you like to stay?
   320→                    </h2>
   321→                    <p className="mt-1 text-sm text-muted-foreground">
   322→                      Choose your dates and party size to see available residences.
   323→                    </p>
   324→                  </div>
   325→
   326→                  <div className="mt-8 grid gap-4 sm:grid-cols-2">
   327→                    <div className="space-y-1.5">
   328→                      <Label htmlFor="bf-checkin" className="text-xs uppercase tracking-wider text-muted-foreground">
   329→                        Check-in
   330→                      </Label>
   331→                      <div className="relative">
   332→                        <CalendarIcon className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   333→                        <Input
   334→                          id="bf-checkin"
   335→                          type="date"
   336→                          value={checkIn}
   337→                          onChange={(e) => setCheckIn(e.target.value)}
   338→                          className="rounded-xl pl-9"
   339→                        />
   340→                      </div>
   341→                    </div>
   342→                    <div className="space-y-1.5">
   343→                      <Label htmlFor="bf-checkout" className="text-xs uppercase tracking-wider text-muted-foreground">
   344→                        Check-out
   345→                      </Label>
   346→                      <div className="relative">
   347→                        <CalendarIcon className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   348→                        <Input
   349→                          id="bf-checkout"
   350→                          type="date"
   351→                          value={checkOut}
   352→                          onChange={(e) => setCheckOut(e.target.value)}
   353→                          className="rounded-xl pl-9"
   354→                        />
   355→                      </div>
   356→                    </div>
   357→                  </div>
   358→
   359→                  <div className="mt-4 grid gap-4 sm:grid-cols-2">
   360→                    <div className="space-y-1.5">
   361→                      <Label className="text-xs uppercase tracking-wider text-muted-foreground">Adults</Label>
   362→                      <Select value={adults} onValueChange={setAdults}>
   363→                        <SelectTrigger className="w-full rounded-xl">
   364→                          <Users className="h-4 w-4 text-muted-foreground" />
   365→                          <SelectValue />
   366→                        </SelectTrigger>
   367→                        <SelectContent>
   368→                          {[1, 2, 3, 4, 5, 6].map((n) => (
   369→                            <SelectItem key={n} value={String(n)}>
   370→                              {n} {n === 1 ? "Adult" : "Adults"}
   371→                            </SelectItem>
   372→                          ))}
   373→                        </SelectContent>
   374→                      </Select>
   375→                    </div>
   376→                    <div className="space-y-1.5">
   377→                      <Label className="text-xs uppercase tracking-wider text-muted-foreground">Children</Label>
   378→                      <Select value={children} onValueChange={setChildren}>
   379→                        <SelectTrigger className="w-full rounded-xl">
   380→                          <Users className="h-4 w-4 text-muted-foreground" />
   381→                          <SelectValue />
   382→                        </SelectTrigger>
   383→                        <SelectContent>
   384→                          {[0, 1, 2, 3, 4].map((n) => (
   385→                            <SelectItem key={n} value={String(n)}>
   386→                              {n} {n === 1 ? "Child" : "Children"}
   387→                            </SelectItem>
   388→                          ))}
   389→                        </SelectContent>
   390→                      </Select>
   391→                    </div>
   392→                  </div>
   393→
   394→                  <Button
   395→                    onClick={onSearchAvailability}
   396→                    size="lg"
   397→                    className="mt-8 w-full rounded-full"
   398→                  >
   399→                    <Search className="h-4 w-4" />
   400→                    Search availability
   401→                  </Button>
   402→
   403→                  <p className="mt-4 text-center text-xs text-muted-foreground">
   404→                    No payment required to hold your reservation — our concierge will reach out
   405→                    within 24 hours to confirm details.
   406→                  </p>
   407→                </Card>
   408→              </motion.div>
   409→            )}
   410→
   411→            {/* STEP 2: Select Room */}
   412→            {step === 2 && (
   413→              <motion.div
   414→                key="step2"
   415→                initial={{ opacity: 0, y: 10 }}
   416→                animate={{ opacity: 1, y: 0 }}
   417→                exit={{ opacity: 0, y: -10 }}
   418→                transition={{ duration: 0.3 }}
   419→              >
   420→                <div className="mb-6 flex flex-wrap items-center justify-between gap-4">
   421→                  <div>
   422→                    <h2 className="font-display text-2xl font-semibold tracking-tight">
   423→                      Available residences
   424→                    </h2>
   425→                    <p className="mt-1 text-sm text-muted-foreground">
   426→                      {formatDate(checkIn)} → {formatDate(checkOut)} · {nights} night{nights > 1 ? "s" : ""} ·{" "}
   427→                      {parseInt(adults, 10) + parseInt(children, 10)} guest
   428→                      {parseInt(adults, 10) + parseInt(children, 10) > 1 ? "s" : ""}
   429→                    </p>
   430→                  </div>
   431→                  <Button variant="outline" onClick={() => setStep(1)} className="rounded-full">
   432→                    <CalendarIcon className="h-4 w-4" />
   433→                    Edit dates
   434→                  </Button>
   435→                </div>
   436→
   437→                {availabilityQuery.isLoading ? (
   438→                  <div className="flex items-center justify-center gap-3 py-20 text-muted-foreground">
   439→                    <Loader2 className="h-5 w-5 animate-spin" />
   440→                    Checking availability…
   441→                  </div>
   442→                ) : availabilityQuery.data ? (
   443→                  <>
   444→                    {availabilityQuery.data.available.length === 0 ? (
   445→                      <Card className="rounded-2xl border-dashed py-12 text-center">
   446→                        <p className="font-medium">No residences available for these dates</p>
   447→                        <p className="mt-1 text-sm text-muted-foreground">
   448→                          Try adjusting your dates or party size.
   449→                        </p>
   450→                        <Button onClick={() => setStep(1)} className="mt-4 rounded-full">
   451→                          Edit search
   452→                        </Button>
   453→                      </Card>
   454→                    ) : (
   455→                      <>
   456→                        <p className="mb-4 text-sm text-muted-foreground">
   457→                          {availabilityQuery.data.available.length} residence
   458→                          {availabilityQuery.data.available.length === 1 ? "" : "s"} available
   459→                        </p>
   460→                        <div className="grid gap-6 sm:grid-cols-2 lg:grid-cols-3">
   461→                          {availabilityQuery.data.available.map((room) => (
   462→                            <RoomCard
   463→                              key={room.id}
   464→                              room={room}
   465→                              nights={nights}
   466→                              totalPrice={room.totalPrice}
   467→                              isAvailable
   468→                              selected={room.id === selectedRoomId}
   469→                              showBookButton={false}
   470→                              showSelectButton
   471→                              onSelect={(r) => onSelectRoom(r as AvailableRoom)}
   472→                              onDetails={(r) =>
   473→                                navigate("room-details", { roomId: r.id })
   474→                              }
   475→                            />
   476→                          ))}
   477→                        </div>
   478→                      </>
   479→                    )}
   480→
   481→                    {availabilityQuery.data.unavailable.length > 0 && (
   482→                      <div className="mt-12">
   483→                        <h3 className="mb-4 text-sm font-medium uppercase tracking-wider text-muted-foreground">
   484→                          Currently unavailable
   485→                        </h3>
   486→                        <div className="grid gap-6 sm:grid-cols-2 lg:grid-cols-3">
   487→                          {availabilityQuery.data.unavailable.slice(0, 6).map((room) => (
   488→                            <RoomCard
   489→                              key={room.id}
   490→                              room={room}
   491→                              nights={nights}
   492→                              totalPrice={room.totalPrice}
   493→                              isAvailable={false}
   494→                              showBookButton={false}
   495→                              showSelectButton
   496→                              onSelect={() =>
   497→                                toast.error("This room is unavailable for your dates")
   498→                              }
   499→                              onDetails={(r) =>
   500→                                navigate("room-details", { roomId: r.id })
   501→                              }
   502→                            />
   503→                          ))}
   504→                        </div>
   505→                      </div>
   506→                    )}
   507→                  </>
   508→                ) : null}
   509→              </motion.div>
   510→            )}
   511→
   512→            {/* STEP 3: Guest Information */}
   513→            {step === 3 && (
   514→              <GuestInfoStep
   515→                key="step3"
   516→                defaultValues={guestInfo}
   517→                onSubmit={onGuestInfoSubmit}
   518→                onBack={() => setStep(2)}
   519→              />
   520→            )}
   521→
   522→            {/* STEP 4: Review & Confirm */}
   523→            {step === 4 && (
   524→              <motion.div
   525→                key="step4"
   526→                initial={{ opacity: 0, y: 10 }}
   527→                animate={{ opacity: 1, y: 0 }}
   528→                exit={{ opacity: 0, y: -10 }}
   529→                transition={{ duration: 0.3 }}
   530→                className="mx-auto max-w-4xl"
   531→              >
   532→                <ReviewStep
   533→                  checkIn={checkIn}
   534→                  checkOut={checkOut}
   535→                  adults={parseInt(adults, 10)}
   536→                  childCount={parseInt(children, 10)}
   537→                  nights={nights}
   538→                  room={selectedRoom}
   539→                  guest={guestInfo}
   540→                  onBack={() => setStep(3)}
   541→                  onConfirm={onConfirmReservation}
   542→                  submitting={createMutation.isPending}
   543→                />
   544→              </motion.div>
   545→            )}
   546→
   547→            {/* STEP 5: Confirmation */}
   548→            {step === 5 && (
   549→              <ConfirmationStep
   550→                reservation={confirmedReservation}
   551→                onCreateAnother={restart}
   552→              />
   553→            )}
   554→          </AnimatePresence>
   555→        </div>
   556→      </section>
   557→    </div>
   558→  );
   559→}

   561→// ============================================================
   562→// Step 3: Guest Info Form
   563→// ============================================================
   564→function GuestInfoStep({
   565→  defaultValues,
   566→  onSubmit,
   567→  onBack,
   568→}: {
   569→  defaultValues: GuestInfoInput | null;
   570→  onSubmit: (values: GuestInfoInput) => void;
   571→  onBack: () => void;
   572→}) {
   573→  const form = useForm<GuestInfoInput>({
   574→    resolver: zodResolver(guestInfoSchema),
   575→    defaultValues: defaultValues ?? {
   576→      firstName: "",
   577→      lastName: "",
   578→      email: "",
   579→      phone: "",
   580→      address: "",
   581→      city: "",
   582→      country: "",
   583→      specialRequests: "",
   584→    },
   585→    mode: "onBlur",
   586→  });
   587→
   588→  return (
   589→    <motion.div
   590→      initial={{ opacity: 0, y: 10 }}
   591→      animate={{ opacity: 1, y: 0 }}
   592→      exit={{ opacity: 0, y: -10 }}
   593→      transition={{ duration: 0.3 }}
   594→      className="mx-auto max-w-3xl"
