   369→                  onView={() => setDetailsId(r.id)}
   370→                  onMutate={(s) =>
   371→                    statusMutation.mutate({ id: r.id, status: s })
   372→                  }
   373→                  pending={statusMutation.isPending}

   376→              </div>

   379→        )}

   382→      {/* Details dialog */}

   384→        <ReservationDetailsDialog

   387→          onClose={() => setDetailsId(null)}
   388→          onMutate={(s) => {
   389→            statusMutation.mutate({ id: selected.id, status: s });

   391→          }}
   392→          pending={statusMutation.isPending}
   393→        />
   394→      )}
   395→
   396→      {/* Create dialog */}

   398→        open={creating}
   399→        onClose={() => setCreating(false)}
   400→        onCreated={() => {
   401→          setCreating(false);
   402→          qc.invalidateQueries({ queryKey: ["admin-reservations"] });
   403→          qc.invalidateQueries({ queryKey: ["admin-dashboard"] });
   404→        }}
   405→      />
   406→    </AdminLayout>
   407→  );
   408→}
   409→
   410→function RowActions({
   411→  reservation,
   412→  onView,
   413→  onMutate,
   414→  pending,
   415→  fullWidth,
   416→}: {
   417→  reservation: Reservation;
   418→  onView: () => void;
   419→  onMutate: (status: BookingStatus) => void;
   420→  pending: boolean;
   421→  fullWidth?: boolean;
   422→}) {
   423→  const s = reservation.status as BookingStatus;
   424→  return (
   425→    <div
   426→      className={cn(
   427→        "flex items-center justify-end gap-1.5",
   428→        fullWidth && "w-full"
   429→      )}
   430→    >
   431→      {s === "PENDING" && (
   432→        <>
   433→          <Button
   434→            size="sm"
   435→            className="h-8 bg-emerald-600 text-white hover:bg-emerald-700"
   436→            disabled={pending}
   437→            onClick={() => onMutate("CONFIRMED")}
   438→          >
   439→            <Check className="size-3.5" />
   440→            Approve
   441→          </Button>
   442→          <Button
   443→            size="sm"
   444→            variant="outline"
   445→            className="h-8 border-red-300 text-red-700 hover:bg-red-50"
   446→            disabled={pending}
   447→            onClick={() => onMutate("REJECTED")}
   448→          >
   449→            <X className="size-3.5" />
   450→            Decline
   451→          </Button>
   452→        </>
   453→      )}
   454→      {s === "CONFIRMED" && (
   455→        <Button
   456→          size="sm"
   457→          className="h-8 bg-primary text-primary-foreground hover:bg-primary/90"
   458→          disabled={pending}
   459→          onClick={() => onMutate("CHECKED_IN")}
   460→        >
   461→          <LogIn className="size-3.5" />
   462→          Check in
   463→        </Button>
   464→      )}
   465→      {s === "CHECKED_IN" && (
   466→        <Button
   467→          size="sm"
   468→          className="h-8 bg-primary text-primary-foreground hover:bg-primary/90"
   469→          disabled={pending}
   470→          onClick={() => onMutate("COMPLETED")}
   471→        >
   472→          <LogOut className="size-3.5" />
   473→          Check out
   474→        </Button>
   475→      )}
   476→      {(s === "COMPLETED" ||
   477→        s === "CANCELLED" ||
   478→        s === "REJECTED" ||
   479→        s === "NO_SHOW") && (
   480→        <Button
   481→          size="sm"
   482→          variant="ghost"
   483→          className="h-8 text-foreground hover:bg-muted"
   484→          onClick={onView}
   485→        >
   486→          <Eye className="size-3.5" />
   487→          View
   488→        </Button>
   489→      )}
   490→      <Button
   491→        size="sm"
   492→        variant="ghost"
   493→        className={cn("h-8 text-foreground hover:bg-muted", fullWidth && "flex-1")}
   494→        onClick={onView}
   495→      >
   496→        <Eye className="size-3.5" />
   497→        Details
   498→      </Button>
   499→    </div>
   500→  );
   501→}

   587→                {reservation.rooms?.[0]?.room?.name ?? "—"}
   588→              </dd>
   589→            </dl>
   590→          </Section>
   591→
   592→          {/* Price */}
   593→          <Section title="Price breakdown">
   594→            <dl className="space-y-1.5 text-sm">
   595→              <div className="flex justify-between">
   596→                <dt className="text-muted-foreground">
   597→                  {reservation.rooms?.[0]?.pricePerNight != null
   598→                    ? formatCurrency(reservation.rooms[0].pricePerNight)
   599→                    : "—"}{" "}
   600→                  × {reservation.nights} nights
   601→                </dt>
   602→                <dd className="text-foreground">
   603→                  {formatCurrency(reservation.totalAmount)}
   604→                </dd>
   605→              </div>
   606→              <div className="flex justify-between border-t border-border pt-1.5 text-base font-medium">
   607→                <dt>Total</dt>
   608→                <dd>{formatCurrency(reservation.totalAmount)}</dd>
   609→              </div>
   610→            </dl>
   611→          </Section>
   612→
   613→          {reservation.specialRequests && (
   614→            <Section title="Special requests">
   615→              <p className="rounded-lg bg-muted/50 p-3 text-sm text-foreground">
   616→                {reservation.specialRequests}
   617→              </p>
   618→            </Section>
   619→          )}
   620→
   621→          {/* Status timeline */}
   622→          <Section title="Timeline">
   623→            <ul className="space-y-2 text-xs">
   624→              <TimelineItem
   625→                label="Booking received"
   626→                date={reservation.createdAt}
   627→                done
   628→              />
   629→              <TimelineItem
   630→                label="Confirmed"
   631→                date={reservation.confirmedAt}
   632→                done={!!reservation.confirmedAt}
   633→              />
   634→              <TimelineItem
   635→                label="Checked in"
   636→                date={reservation.checkedInAt}
   637→                done={!!reservation.checkedInAt}
   638→              />
   639→              <TimelineItem
   640→                label="Checked out"
   641→                date={reservation.checkedOutAt}
   642→                done={!!reservation.checkedOutAt}
   643→              />
   644→            </ul>
   645→          </Section>
   646→        </div>
   647→
   648→        <DialogFooter className="flex-row flex-wrap gap-2 border-t border-border px-4 py-4 sm:px-6">
   649→          {s === "PENDING" && (
   650→            <>
   651→              <Button
   652→                variant="outline"
   653→                className="w-full border-red-300 text-red-700 hover:bg-red-50 sm:w-auto"
   654→                disabled={pending}
   655→                onClick={() => onMutate("REJECTED")}
   656→              >
   657→                <UserX className="size-4" />
   658→                Decline
   659→              </Button>
   660→              <Button
   661→                className="ml-auto w-full bg-emerald-600 text-white hover:bg-emerald-700 sm:w-auto"
   662→                disabled={pending}
   663→                onClick={() => onMutate("CONFIRMED")}
   664→              >
   665→                <Check className="size-4" />
   666→                Approve
   667→              </Button>
   668→            </>
   669→          )}
   670→          {s === "CONFIRMED" && (
   671→            <Button
   672→              className="ml-auto w-full bg-primary text-primary-foreground hover:bg-primary/90 sm:w-auto"
   673→              disabled={pending}
   674→              onClick={() => onMutate("CHECKED_IN")}
   675→            >
   676→              <LogIn className="size-4" />
   677→              Check in
   678→            </Button>
   679→          )}
   680→          {s === "CHECKED_IN" && (
   681→            <Button
   682→              className="ml-auto w-full bg-primary text-primary-foreground hover:bg-primary/90 sm:w-auto"
   683→              disabled={pending}
   684→              onClick={() => onMutate("COMPLETED")}
   685→            >
   686→              <LogOut className="size-4" />
   687→              Check out
   688→            </Button>
   689→          )}
   690→          {(s === "COMPLETED" ||
   691→            s === "CANCELLED" ||
   692→            s === "REJECTED" ||
   693→            s === "NO_SHOW") && (
   694→            <Button variant="outline" className="ml-auto w-full sm:w-auto" onClick={onClose}>
   695→              Close
   696→            </Button>
   697→          )}
   698→        </DialogFooter>
   699→      </DialogContent>
   700→    </Dialog>
   701→  );
   702→}

   753→function CreateReservationDialog({
   754→  open,
   755→  onClose,
   756→  onCreated,
   757→}: {
   758→  open: boolean;
   759→  onClose: () => void;
   760→  onCreated: () => void;
   761→}) {
   762→  const qc = useQueryClient();
   763→  const { data: roomsData } = useQuery({
   764→    queryKey: ["admin-rooms", "active"],
   765→    queryFn: () => apiFetch<{ rooms: Room[] }>("/api/rooms"),
   766→    enabled: open,
   767→  });
   768→  const rooms = (roomsData?.rooms ?? []).filter((r) => r.isActive);
   769→
   770→  const {
   771→    register,
   772→    handleSubmit,
   773→    watch,
   774→    setValue,
   775→    reset,
   776→    formState: { errors },
   777→  } = useForm<ReservationCreateInput>({
   778→    resolver: zodResolver(reservationCreateSchema),
   779→    defaultValues: {
   780→      roomId: "",
   781→      checkIn: "",
   782→      checkOut: "",
   783→      adults: 2,
   784→      children: 0,
   785→      guest: {
   786→        firstName: "",
   787→        lastName: "",
   788→        email: "",
   789→        phone: "",
   790→        address: "",
   791→        city: "",
   792→        country: "",
   793→      },
   794→      specialRequests: "",
   795→    },
   796→  });
   797→
   798→  const roomId = watch("roomId");
   799→  const checkIn = watch("checkIn");
   800→  const checkOut = watch("checkOut");
   801→  const selectedRoom = rooms.find((r) => r.id === roomId);
   802→  const nights =
   803→    checkIn && checkOut ? nightsBetween(checkIn, checkOut) : 0;
   804→  const total = selectedRoom ? selectedRoom.pricePerNight * nights : 0;
   805→
   806→  const mutation = useMutation({
   807→    mutationFn: (values: ReservationCreateInput) =>
   808→      apiFetch<{ reservation: Reservation }>("/api/reservations", {
   809→        method: "POST",
   810→        body: JSON.stringify({ ...values, source: "WALK_IN" }),
   811→      }),
   812→    onSuccess: (data) => {
   813→      toast.success(
   814→        `Reservation ${data.reservation.referenceNo} created.`
   815→      );
   816→      qc.invalidateQueries({ queryKey: ["admin-reservations"] });
   817→      qc.invalidateQueries({ queryKey: ["admin-dashboard"] });
   818→      reset();
   819→      onCreated();
   820→    },
   821→    onError: (err) => {
   822→      const message =
   823→        err instanceof ApiError
   824→          ? err.message
   825→          : "Couldn't create reservation.";
   826→      toast.error(message);
   827→    },
   828→  });
   829→
   830→  return (
   831→    <Dialog
   832→      open={open}
   833→      onOpenChange={(o) => {
   834→        if (!o) {
   835→          reset();
   836→          onClose();
   837→        }
   838→      }}
   839→    >
   840→      <DialogContent className="max-h-[90vh] max-w-2xl gap-0 overflow-hidden p-0">
   841→        <DialogHeader className="border-b border-border px-4 py-5 sm:px-6">
   842→          <DialogTitle className="font-display text-xl font-medium tracking-tight">
   843→            New reservation
   844→          </DialogTitle>
   845→          <DialogDescription>
   846→            Create a walk-in or phone booking. Status starts as Pending.
   847→          </DialogDescription>
   848→        </DialogHeader>
   849→
   850→        <form
   851→          onSubmit={handleSubmit((v) => mutation.mutate(v))}
   852→          className="flex max-h-[70vh] flex-col"
   853→        >
   854→          <div className="flex-1 overflow-y-auto px-4 py-5 sm:px-6">
   855→            {/* Guest */}
   856→            <div className="mb-5">
   857→              <h4 className="mb-3 text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   858→                Guest information
   859→              </h4>
   860→              <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   861→                <Field label="First name" error={errors.guest?.firstName?.message}>
   862→                  <Input {...register("guest.firstName")} className="h-10" />
   863→                </Field>
   864→                <Field label="Last name" error={errors.guest?.lastName?.message}>
   865→                  <Input {...register("guest.lastName")} className="h-10" />
   866→                </Field>
   867→                <Field label="Email" error={errors.guest?.email?.message}>
   868→                  <Input
   869→                    type="email"
   870→                    {...register("guest.email")}
   871→                    className="h-10"
   872→                  />
   873→                </Field>
   874→                <Field label="Phone" error={errors.guest?.phone?.message}>
   875→                  <Input {...register("guest.phone")} className="h-10" />
   876→                </Field>
   877→                <Field label="City" error={errors.guest?.city?.message}>
   878→                  <Input {...register("guest.city")} className="h-10" />
   879→                </Field>
   880→                <Field label="Country" error={errors.guest?.country?.message}>
   881→                  <Input {...register("guest.country")} className="h-10" />
   882→                </Field>
   883→              </div>
   884→            </div>
   885→
   886→            {/* Stay */}
   887→            <div className="mb-5">
   888→              <h4 className="mb-3 text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   889→                Stay details
   890→              </h4>
   891→              <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   892→                <Field label="Room" error={errors.roomId?.message}>
   893→                  <Select
   894→                    value={roomId}
   895→                    onValueChange={(v) => setValue("roomId", v)}
   896→                  >
   897→                    <SelectTrigger className="h-10">
   898→                      <SelectValue placeholder="Select a room" />
   899→                    </SelectTrigger>
   900→                    <SelectContent>
   901→                      {rooms.map((r) => (
   902→                        <SelectItem key={r.id} value={r.id}>
   903→                          {r.name} · {formatCurrency(r.pricePerNight)}/night
   904→                        </SelectItem>
   905→                      ))}
   906→                    </SelectContent>
   907→                  </Select>
   908→                </Field>
   909→                <div className="grid grid-cols-2 gap-3">
   910→                  <Field label="Adults" error={errors.adults?.message}>
   911→                    <Input
   912→                      type="number"
   913→                      min={1}
   914→                      {...register("adults", { valueAsNumber: true })}
   915→                      className="h-10"
   916→                    />
   917→                  </Field>
   918→                  <Field label="Children" error={errors.children?.message}>
   919→                    <Input
   920→                      type="number"
   921→                      min={0}
   922→                      {...register("children", { valueAsNumber: true })}
   923→                      className="h-10"
   924→                    />
   925→                  </Field>
   926→                </div>
   927→                <Field label="Check in" error={errors.checkIn?.message}>
   928→                  <Input
   929→                    type="date"
   930→                    {...register("checkIn")}
   931→                    className="h-10"
   932→                  />
   933→                </Field>
   934→                <Field label="Check out" error={errors.checkOut?.message}>
   935→                  <Input
   936→                    type="date"
   937→                    {...register("checkOut")}
   938→                    className="h-10"
   939→                  />
   940→                </Field>
   941→              </div>
   942→            </div>
   943→
   944→            {/* Special requests */}
   945→            <Field
   946→              label="Special requests (optional)"
   947→              error={errors.specialRequests?.message}
   948→            >
   949→              <Textarea
   950→                {...register("specialRequests")}
   951→                rows={2}
   952→                placeholder="Late check-in, birthday cake, extra towels…"
   953→              />
   954→            </Field>
   955→
   956→            {/* Price summary */}
   957→            {selectedRoom && nights > 0 && (
   958→              <div className="mt-5 rounded-lg border border-border bg-sand/40 p-4">
   959→                <div className="flex items-center justify-between text-sm">
   960→                  <span className="text-muted-foreground">
   961→                    {formatCurrency(selectedRoom.pricePerNight)} × {nights}{" "}
   962→                    {nights === 1 ? "night" : "nights"}
   963→                  </span>
   964→                  <span className="font-medium text-foreground">
   965→                    {formatCurrency(total)}
   966→                  </span>
   967→                </div>
   968→                <div className="mt-1.5 flex items-center justify-between border-t border-border pt-1.5 text-base font-medium">
   969→                  <span>Total</span>
   970→                  <span>{formatCurrency(total)}</span>
   971→                </div>
   972→              </div>
   973→            )}
   974→          </div>
   975→
   976→          <DialogFooter className="flex-row flex-wrap gap-2 border-t border-border px-4 py-4 sm:px-6">
   977→            <Button
   978→              type="button"
   979→              variant="outline"
   980→              className="w-full sm:w-auto"
   981→              onClick={() => {
   982→                reset();
   983→                onClose();
   984→              }}
   985→            >
   986→              Cancel
   987→            </Button>
   988→            <Button
   989→              type="submit"
   990→              className="w-full bg-primary text-primary-foreground hover:bg-primary/90 sm:w-auto"
   991→              disabled={mutation.isPending}
   992→            >
   993→              {mutation.isPending ? "Creating…" : "Create reservation"}
   994→            </Button>
   995→          </DialogFooter>
   996→        </form>
   997→      </DialogContent>
   998→    </Dialog>
   999→  );
  1000→}
