    60→type Step = 1 | 2 | 3;
    61→
    62→const STEPS: { value: Step; label: string }[] = [
    63→  { value: 1, label: "Dates" },
    64→  { value: 2, label: "Details" },
    65→  { value: 3, label: "Confirm" },

    67→
    68→export function BookingFlow() {
    69→  const navigate = useViewStore((s) => s.navigate);
    70→  const booking = useBookingStore();
    71→  const setSearch = useBookingStore((s) => s.setSearch);
    72→  const selectRoom = useBookingStore((s) => s.selectRoom);
    73→
    74→  const [step, setStep] = React.useState<Step>(1);
    75→  const [confirmed, setConfirmed] = React.useState<Reservation | null>(null);
    76→  // Guest details captured in Step 2 — kept in local state because the booking
    77→  // store only holds dates/guests/room (per the lead agent's foundation design).
    78→  const [guestInfo, setGuestInfo] = React.useState<GuestInfoInput>({
    79→    firstName: "",
    80→    lastName: "",
    81→    email: "",
    82→    phone: "",
    83→    city: "",
    84→    country: "",
    85→    specialRequests: "",
    86→  });
    87→
    88→  const {
    89→    data: roomsData,
    90→    isLoading: roomsLoading,
    91→  } = useQuery({
    92→    queryKey: ["rooms", "booking-flow"],
    93→    queryFn: () => apiFetch<RoomsResponse>("/api/rooms"),
    94→  });
    95→  const rooms = roomsData?.rooms ?? [];
    96→  const selectedRoom = rooms.find((r) => r.id === booking.selectedRoomId);
    97→
    98→  const createMutation = useMutation({
    99→    mutationFn: (payload: {
   100→      roomId: string;
   101→      checkIn: string;
   102→      checkOut: string;
   103→      adults: number;
   104→      children: number;
   105→      guest: GuestInfoInput;
   106→      specialRequests?: string;
   107→    }) =>
   108→      apiFetch<ReservationResponse>("/api/reservations", {
   109→        method: "POST",
   110→        body: JSON.stringify(payload),
   111→      }),
   112→    onSuccess: (data) => {
   113→      setConfirmed(data.reservation);
   114→      setStep(3);
   115→      toast.success("Booking request received!");
   116→    },
   117→    onError: (err) => {
   118→      const msg =
   119→        err instanceof ApiError
   120→          ? err.message
   121→          : "Something went wrong. Please try again.";
   122→      toast.error(msg);
   123→    },
   124→  });
   125→
   126→  // Reset confirmation when leaving the flow.
   127→  React.useEffect(() => {
   128→    return () => {
   129→      if (confirmed) {
   130→        // If we have a confirmed booking, reset store on unmount so a fresh
   131→        // visit starts at step 1.
   132→        booking.reset();
   133→      }
   134→    };
   135→  }, [confirmed, booking]);
   136→
   137→  // ============== Confirmation screen ==============
   138→  if (confirmed) {
   139→    return (
   140→      <ConfirmationScreen
   141→        reservation={confirmed}
   142→        onReset={() => {
   143→          booking.reset();
   144→          setConfirmed(null);
   145→          setStep(1);
   146→          navigate("home");
   147→        }}
   148→      />
   149→    );
   150→  }
   151→
   152→  return (
   153→    <div className="pt-8 sm:pt-12">
   154→      {/* Progress indicator */}
   155→      <div className="border-b border-border bg-section">
   156→        <div className="container-luxury py-5">
   157→          <div className="flex items-center justify-center gap-2 sm:gap-4">
   158→            {STEPS.map((s, i) => {
   159→              const active = step === s.value;
   160→              const done = step > s.value;
   161→              return (
   162→                <React.Fragment key={s.value}>
   163→                  <button
   164→                    type="button"
   165→                    onClick={() => {
   166→                      // Allow going back to previous steps.
   167→                      if (s.value < step) setStep(s.value);
   168→                    }}
   169→                    disabled={s.value > step}
   170→                    className="flex items-center gap-2"
   171→                  >
   172→                    <span
   173→                      className={cn(
   174→                        "flex h-8 w-8 items-center justify-center rounded-full text-sm font-semibold transition-colors",
   175→                        active
   176→                          ? "bg-primary text-white"
   177→                          : done
   178→                            ? "bg-primary/15 text-primary"
   179→                            : "bg-muted text-muted-foreground"
   180→                      )}
   181→                    >
   182→                      {done ? <Check className="h-4 w-4" /> : s.value}
   183→                    </span>
   184→                    <span
   185→                      className={cn(
   186→                        "text-sm font-medium transition-colors",
   187→                        active
   188→                          ? "text-foreground"
   189→                          : "text-muted-foreground"
   190→                      )}
   191→                    >
   192→                      {s.label}
   193→                    </span>
   194→                  </button>
   195→                  {i < STEPS.length - 1 && (
   196→                    <span
   197→                      className={cn(
   198→                        "h-px w-8 sm:w-16",
   199→                        step > s.value ? "bg-primary/40" : "bg-border"
   200→                      )}
   201→                    />
   202→                  )}
   203→                </React.Fragment>
   204→              );
   205→            })}
   206→          </div>
   207→        </div>
   208→      </div>
   209→
   210→      <div className="py-8 sm:py-12">
   211→        <div className="container-luxury">
   212→          {/* ============================================================
   213→              STEP 1 — Dates & Guests + Choose your stay
   214→          ============================================================ */}
   215→          {step === 1 && (
   216→            <Step1Dates
   217→              booking={booking}
   218→              setSearch={setSearch}
   219→              selectRoom={selectRoom}
   220→              rooms={rooms}
   221→              roomsLoading={roomsLoading}
   222→              selectedRoom={selectedRoom}
   223→              onContinue={() => setStep(2)}
   224→            />
   225→          )}
   226→
   227→          {/* ============================================================
   228→              STEP 2 — Your Details
   229→          ============================================================ */}
   230→          {step === 2 && (
   231→            <Step2Details
   232→              defaultValues={guestInfo}
   233→              onBack={() => setStep(1)}
   234→              onContinue={(values) => {
   235→                setGuestInfo(values);
   236→                setStep(3);
   237→              }}
   238→            />
   239→          )}
   240→
   241→          {/* ============================================================
   242→              STEP 3 — Review & Confirm
   243→          ============================================================ */}
   244→          {step === 3 && (
   245→            <Step3Confirm
   246→              booking={booking}
   247→              guestInfo={guestInfo}
   248→              selectedRoom={selectedRoom}
   249→              onBack={() => setStep(2)}
   250→              onConfirm={() => {
   251→                if (!selectedRoom) {
   252→                  toast.error("Please choose your stay first");
   253→                  setStep(1);
   254→                  return;
   255→                }
   256→                createMutation.mutate({
   257→                  roomId: selectedRoom.id,
   258→                  checkIn: booking.checkIn,
   259→                  checkOut: booking.checkOut,
   260→                  adults: booking.adults,
   261→                  children: booking.children,
   262→                  guest: guestInfo,
   263→                  specialRequests: guestInfo.specialRequests || undefined,
   264→                });
   265→              }}
   266→              submitting={createMutation.isPending}
   267→            />
   268→          )}
   269→        </div>
   270→      </div>
   271→    </div>
   272→  );
   273→}

   278→function Step1Dates({
   279→  booking,
   280→  setSearch,
   281→  selectRoom,
   282→  rooms,
   283→  roomsLoading,
   284→  selectedRoom,
   285→  onContinue,
   286→}: {
   287→  booking: ReturnType<typeof useBookingStore.getState>;
   288→  setSearch: (patch: Partial<{ checkIn: string; checkOut: string; adults: number; children: number; selectedRoomId: string | null }>) => void;
   289→  selectRoom: (id: string) => void;
   290→  rooms: Room[];
   291→  roomsLoading: boolean;
   292→  selectedRoom?: Room;
   293→  onContinue: () => void;
   294→}) {
   295→  const checkIn = booking.checkIn || defaultDate(1);
   296→  const checkOut = booking.checkOut || defaultDate(3);
   297→  const nights = nightsBetween(checkIn, checkOut);
   298→
   299→  const onContinueClick = () => {
   300→    if (!booking.checkIn || !booking.checkOut) {
   301→      setSearch({ checkIn, checkOut });
   302→    }
   303→    if (!selectedRoom) {
   304→      toast.error("Please pick your stay below first");
   305→      return;
   306→    }
   307→    onContinue();
   308→  };
   309→
   310→  return (
   311→    <div className="mx-auto max-w-4xl">
   312→      <FadeUpSection>
   313→        <h1 className="font-display text-3xl font-semibold tracking-tight sm:text-4xl">
   314→          Pick your dates
   315→        </h1>
   316→        <p className="mt-2 text-muted-foreground">
   317→          Tell us when you&rsquo;d like to visit and how many guests. Then
   318→          choose the whole villa or a single bedroom.
   319→        </p>
   320→      </FadeUpSection>
   321→
   322→      {/* Dates + Guests card */}
   323→      <FadeUpSection delay={0.05} className="mt-6">
   324→        <Card className="rounded-xl border border-border bg-card p-5 shadow-card sm:p-6">
   325→          <div className="grid gap-4 sm:grid-cols-2">
   326→            <div className="space-y-1.5">
   327→              <Label
   328→                htmlFor="bk-checkin"
   329→                className="text-xs font-medium uppercase tracking-wider text-muted-foreground"
   330→              >
   331→                Check-in
   332→              </Label>
   333→              <div className="relative">
   334→                <CalendarIcon className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-primary/60" />
   335→                <Input
   336→                  id="bk-checkin"
   337→                  type="date"
   338→                  value={checkIn}
   339→                  min={defaultDate(0)}
   340→                  onChange={(e) =>
   341→                    setSearch({ checkIn: e.target.value })
   342→                  }
   343→                  className="rounded-lg pl-9"
   344→                />
   345→              </div>
   346→            </div>
   347→            <div className="space-y-1.5">
   348→              <Label
   349→                htmlFor="bk-checkout"
   350→                className="text-xs font-medium uppercase tracking-wider text-muted-foreground"
   351→              >
   352→                Check-out
   353→              </Label>
   354→              <div className="relative">
   355→                <CalendarIcon className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-primary/60" />
   356→                <Input
   357→                  id="bk-checkout"
   358→                  type="date"
   359→                  value={checkOut}
   360→                  min={checkIn || defaultDate(1)}
   361→                  onChange={(e) =>
   362→                    setSearch({ checkOut: e.target.value })
   363→                  }
   364→                  className="rounded-lg pl-9"
   365→                />
   366→              </div>
   367→            </div>
   368→          </div>
   369→
   370→          {/* Guests steppers */}
   371→          <div className="mt-5 grid gap-4 sm:grid-cols-2">
   372→            <Stepper
   373→              label="Adults"
   374→              hint="Ages 13+"
   375→              value={booking.adults}
   376→              min={1}
   377→              max={25}
   378→              onChange={(v) => setSearch({ adults: v })}
   379→            />
   380→            <Stepper
   381→              label="Children"
   382→              hint="Ages 0–12"
   383→              value={booking.children}
   384→              min={0}
   385→              max={20}
   386→              onChange={(v) => setSearch({ children: v })}
   387→            />
   388→          </div>
   389→
   390→          {/* Summary */}
   391→          <div className="mt-5 flex flex-wrap items-center gap-x-6 gap-y-2 rounded-lg bg-section px-4 py-3 text-sm">
   392→            <span className="inline-flex items-center gap-1.5 text-foreground">
   393→              <CalendarIcon className="h-3.5 w-3.5 text-primary" />
   394→              {formatDate(checkIn)} → {formatDate(checkOut)}
   395→            </span>
   396→            <span className="inline-flex items-center gap-1.5 text-foreground">
   397→              <Users className="h-3.5 w-3.5 text-primary" />
   398→              {booking.adults + booking.children} guest
   399→              {booking.adults + booking.children === 1 ? "" : "s"}
   400→            </span>
   401→            <span className="text-muted-foreground">
   402→              {nights} night{nights === 1 ? "" : "s"}
   403→            </span>
   404→          </div>
   405→        </Card>
   406→      </FadeUpSection>
   407→
   408→      {/* Selected room summary OR room picker */}
   409→      {selectedRoom ? (
   410→        <FadeUpSection delay={0.1} className="mt-8">
   411→          <div className="flex items-center justify-between">
   412→            <h2 className="font-display text-2xl font-semibold tracking-tight">
   413→              Your stay
   414→            </h2>
   415→            <button
   416→              onClick={() => selectRoom("")}
   417→              className="text-sm font-medium text-primary transition-colors hover:text-coral"
   418→            >
   419→              Change
   420→            </button>
   421→          </div>
   422→          <Card className="mt-4 flex overflow-hidden rounded-xl border border-border bg-card shadow-card">
   423→            <div className="relative aspect-[4/3] w-32 shrink-0 bg-muted sm:w-48">
   424→              {selectedRoom.images?.[0] ? (
   425→                <img
   426→                  src={selectedRoom.images[0].url}
   427→                  alt={selectedRoom.name}
   428→                  className="h-full w-full object-cover"
   429→                />
   430→              ) : (
   431→                <div className="flex h-full w-full items-center justify-center">
   432→                  <BedDouble className="h-8 w-8 text-muted-foreground/40" />
   433→                </div>
   434→              )}
   435→            </div>
   436→            <div className="flex flex-1 flex-col justify-between p-4 sm:p-5">
   437→              <div>
   438→                <p className="eyebrow text-[0.6rem]">
   439→                  {selectedRoom.type?.name}
   440→                </p>
   441→                <h3 className="mt-1 font-display text-lg font-semibold">
   442→                  {selectedRoom.name}
   443→                </h3>
   444→                <p className="mt-1 text-sm text-muted-foreground">
   445→                  Sleeps {selectedRoom.capacity}
   446→                </p>
   447→              </div>
   448→              <div className="mt-3 font-display text-lg font-semibold text-primary">
   449→                {formatCurrency(selectedRoom.pricePerNight)}
   450→                <span className="ml-1 text-xs font-normal text-muted-foreground">
   451→                  / night
   452→                </span>
   453→              </div>
   454→            </div>
   455→          </Card>
   456→        </FadeUpSection>
   457→      ) : (
   458→        <FadeUpSection delay={0.1} className="mt-8">
   459→          <h2 className="font-display text-2xl font-semibold tracking-tight">
   460→            Choose your stay
   461→          </h2>
   462→          <p className="mt-1 text-sm text-muted-foreground">
   463→            Pick the whole villa for a group celebration, or a single bedroom
   464→            for a quieter escape.
   465→          </p>
   466→          <div className="mt-5 grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
   467→            {roomsLoading
   468→              ? Array.from({ length: 5 }).map((_, i) => (
   469→                  <RoomCardSkeleton key={i} />
   470→                ))
   471→              : rooms.map((room) => (
   472→                  <RoomCard
   473→                    key={room.id}
   474→                    room={room}
   475→                    selected={room.id === booking.selectedRoomId}
   476→                    onSelect={(r) => selectRoom(r.id)}
   477→                  />
   478→                ))}
   479→          </div>
   480→        </FadeUpSection>
   481→      )}
   482→
   483→      {/* Mobile sticky continue bar */}
   484→      <MobileStickyBar>
   485→        <Button
   486→          onClick={onContinueClick}
   487→          size="lg"
   488→          className="w-full rounded-full"
   489→        >
   490→          Continue
   491→          <ArrowRight className="h-4 w-4" />
   492→        </Button>
   493→      </MobileStickyBar>
   494→
   495→      {/* Desktop continue */}
   496→      <div className="mt-8 hidden justify-end sm:flex">
   497→        <Button
   498→          onClick={onContinueClick}
   499→          size="lg"
   500→          className="rounded-full"
   501→        >
   502→          Continue
   503→          <ArrowRight className="h-4 w-4" />
   504→        </Button>
   505→      </div>
   506→    </div>
   507→  );
   508→}

  1179→function buildIcs(reservation: Reservation): string {
  1180→  const dt = (s: string) => {
  1181→    const d = new Date(s);
  1182→    // All-day event: check-out day is exclusive.
  1183→    return (
  1184→      d.getUTCFullYear() +
  1185→      String(d.getUTCMonth() + 1).padStart(2, "0") +
  1186→      String(d.getUTCDate()).padStart(2, "0")
  1187→    );
  1188→  };
  1189→  const lines = [
  1190→    "BEGIN:VCALENDAR",
  1191→    "VERSION:2.0",
  1192→    "PRODID:-//The Twenty-Fifth//Booking//EN",
  1193→    "CALSCALE:GREGORIAN",
  1194→    "METHOD:PUBLISH",
  1195→    "BEGIN:VEVENT",
  1196→    `UID:${reservation.referenceNo}@the25thinzambales.com`,
  1197→    `DTSTAMP:${dt(new Date().toISOString())}T000000Z`,
  1198→    `DTSTART;VALUE=DATE:${dt(reservation.checkIn)}`,
  1199→    `DTEND;VALUE=DATE:${dt(reservation.checkOut)}`,
  1200→    `SUMMARY:Stay at The Twenty-Fifth`,
  1201→    `LOCATION:${RESORT_INFO.address}`,
  1202→    `DESCRIPTION:Reference: ${reservation.referenceNo}\\nGuests: ${reservation.adults + reservation.children}\\nStatus: ${reservation.status}`,
  1203→    "END:VEVENT",
  1204→    "END:VCALENDAR",
  1205→  ];
  1206→  return lines.join("\r\n");
  1207→}
