    99→    99→    99→/** Count reservations visible under a given tab (History aggregates 4 statuses). */
   100→   100→   100→function countForTab(tab: string, reservations: Reservation[]): number {
   101→   101→   101→  if (tab === "ALL") return reservations.length;
   102→   102→   102→  if (tab === "HISTORY") {
   103→   103→   103→    return reservations.filter((r) =>
   104→   104→   104→      HISTORY_STATUSES.includes(r.status as BookingStatus)
   105→   105→   105→    ).length;
   106→   106→   106→  }
   107→   107→   107→  return reservations.filter((r) => r.status === tab).length;
   108→   108→   108→}
   109→   109→   109→
   110→   110→   110→/** Build a compact page range with ellipses for large page counts. */
   111→   111→   111→function getPageRange(
   112→   112→   112→  current: number,
   113→   113→   113→  total: number
   114→   114→   114→): (number | "ellipsis")[] {
   115→   115→   115→  if (total <= 7) {
   116→   116→   116→    return Array.from({ length: total }, (_, i) => i + 1);
   117→   117→   117→  }
   118→   118→   118→  if (current <= 4) {
   119→   119→   119→    return [1, 2, 3, 4, 5, "ellipsis", total];
   120→   120→   120→  }
   121→   121→   121→  if (current >= total - 3) {
   122→   122→   122→    return [1, "ellipsis", total - 4, total - 3, total - 2, total - 1, total];
   123→   123→   123→  }
   124→   124→   124→  return [1, "ellipsis", current - 1, current, current + 1, "ellipsis", total];
   125→   125→   125→}
