   370→                  }
   371→                  pending={statusMutation.isPending}

   373→                />
   374→              </div>
   375→            </Card>
   376→          ))
   377→        )}

   380→      {/* Details dialog */}

   386→          onMutate={(s) => {
   387→            statusMutation.mutate({ id: selected.id, status: s });
   388→            setDetailsId(null);
   389→          }}
   390→          pending={statusMutation.isPending}
   391→        />
   392→      )}
   393→
   394→      {/* Create dialog */}
   395→      <CreateReservationDialog
   396→        open={creating}
   397→        onClose={() => setCreating(false)}
   398→        onCreated={() => {
   399→          setCreating(false);
   400→          qc.invalidateQueries({ queryKey: ["admin-reservations"] });
   401→          qc.invalidateQueries({ queryKey: ["admin-dashboard"] });
   402→        }}
   403→      />
   404→    </AdminLayout>
   405→  );

   407→
   408→function RowActions({
   409→  reservation,
   410→  onView,
   411→  onMutate,
   412→  pending,
   413→  fullWidth,
   414→}: {
   415→  reservation: Reservation;
   416→  onView: () => void;
   417→  onMutate: (status: BookingStatus) => void;
   418→  pending: boolean;
   419→  fullWidth?: boolean;
   420→}) {
   421→  const s = reservation.status as BookingStatus;
   422→  return (
   423→    <div
   424→      className={cn(
   425→        "flex items-center justify-end gap-1.5",
   426→        fullWidth && "w-full"
   427→      )}
   428→    >
   429→      {s === "PENDING" && (
   430→        <>
   431→          <Button
   432→            size="sm"
   433→            className="h-8 bg-emerald-600 text-white hover:bg-emerald-700"
   434→            disabled={pending}
   435→            onClick={() => onMutate("CONFIRMED")}
   436→          >
   437→            <Check className="size-3.5" />
   438→            Approve
   439→          </Button>
   440→          <Button
   441→            size="sm"
   442→            variant="outline"
   443→            className="h-8 border-red-300 text-red-700 hover:bg-red-50"
   444→            disabled={pending}
   445→            onClick={() => onMutate("REJECTED")}
   446→          >
   447→            <X className="size-3.5" />
   448→            Decline
   449→          </Button>
   450→        </>
   451→      )}
   452→      {s === "CONFIRMED" && (
   453→        <Button
   454→          size="sm"
   455→          className="h-8 bg-primary text-white hover:bg-primary/90"
   456→          disabled={pending}
   457→          onClick={() => onMutate("CHECKED_IN")}
   458→        >
   459→          <LogIn className="size-3.5" />
   460→          Check in
   461→        </Button>
   462→      )}
   463→      {s === "CHECKED_IN" && (
   464→        <Button
   465→          size="sm"
   466→          className="h-8 bg-primary text-white hover:bg-primary/90"
   467→          disabled={pending}
   468→          onClick={() => onMutate("COMPLETED")}
   469→        >
   470→          <LogOut className="size-3.5" />
   471→          Check out
   472→        </Button>
   473→      )}
   474→      {(s === "COMPLETED" ||
   475→        s === "CANCELLED" ||
   476→        s === "REJECTED" ||
   477→        s === "NO_SHOW") && (
   478→        <Button
   479→          size="sm"
   480→          variant="ghost"
   481→          className="h-8 text-foreground hover:bg-muted"
   482→          onClick={onView}
   483→        >
   484→          <Eye className="size-3.5" />
   485→          View
   486→        </Button>
   487→      )}
   488→      <Button
   489→        size="sm"
   490→        variant="ghost"
   491→        className={cn("h-8 text-foreground hover:bg-muted", fullWidth && "flex-1")}
   492→        onClick={onView}
   493→      >
   494→        <Eye className="size-3.5" />
   495→        Details
   496→      </Button>
   497→    </div>
   498→  );
   499→}

   591→          <Section title="Price breakdown">
   592→            <dl className="space-y-1.5 text-sm">
   593→              <div className="flex justify-between">
   594→                <dt className="text-muted-foreground">
   595→                  {reservation.rooms?.[0]?.pricePerNight != null
   596→                    ? formatCurrency(reservation.rooms[0].pricePerNight)
   597→                    : "—"}{" "}
   598→                  × {reservation.nights} nights
   599→                </dt>
   600→                <dd className="text-foreground">
   601→                  {formatCurrency(reservation.totalAmount)}
   602→                </dd>
   603→              </div>
   604→              <div className="flex justify-between border-t border-border pt-1.5 text-base font-medium">
   605→                <dt>Total</dt>
   606→                <dd>{formatCurrency(reservation.totalAmount)}</dd>
   607→              </div>
   608→            </dl>
   609→          </Section>
   610→
   611→          {reservation.specialRequests && (
   612→            <Section title="Special requests">
   613→              <p className="rounded-lg bg-muted/50 p-3 text-sm text-foreground">
   614→                {reservation.specialRequests}
   615→              </p>
   616→            </Section>
   617→          )}
   618→
   619→          {/* Status timeline */}
   620→          <Section title="Timeline">
   621→            <ul className="space-y-2 text-xs">
   622→              <TimelineItem
   623→                label="Booking received"
   624→                date={reservation.createdAt}
   625→                done
   626→              />
   627→              <TimelineItem
   628→                label="Confirmed"
   629→                date={reservation.confirmedAt}
   630→                done={!!reservation.confirmedAt}
   631→              />
   632→              <TimelineItem
   633→                label="Checked in"
   634→                date={reservation.checkedInAt}
   635→                done={!!reservation.checkedInAt}
   636→              />
   637→              <TimelineItem
   638→                label="Checked out"
   639→                date={reservation.checkedOutAt}
   640→                done={!!reservation.checkedOutAt}
   641→              />
   642→            </ul>
   643→          </Section>
   644→        </div>
   645→
   646→        <DialogFooter className="flex-row flex-wrap gap-2 border-t border-border px-6 py-4">
   647→          {s === "PENDING" && (
   648→            <>
   649→              <Button
   650→                variant="outline"
   651→                className="border-red-300 text-red-700 hover:bg-red-50"
   652→                disabled={pending}
   653→                onClick={() => onMutate("REJECTED")}
   654→              >
   655→                <UserX className="size-4" />
   656→                Decline
   657→              </Button>
   658→              <Button
   659→                className="ml-auto bg-emerald-600 text-white hover:bg-emerald-700"
   660→                disabled={pending}
   661→                onClick={() => onMutate("CONFIRMED")}
   662→              >
   663→                <Check className="size-4" />
   664→                Approve
   665→              </Button>
   666→            </>
   667→          )}
   668→          {s === "CONFIRMED" && (
   669→            <Button
   670→              className="ml-auto bg-primary text-white hover:bg-primary/90"
   671→              disabled={pending}
   672→              onClick={() => onMutate("CHECKED_IN")}
   673→            >
   674→              <LogIn className="size-4" />
   675→              Check in
   676→            </Button>
   677→          )}
   678→          {s === "CHECKED_IN" && (
   679→            <Button
   680→              className="ml-auto bg-primary text-white hover:bg-primary/90"
   681→              disabled={pending}
   682→              onClick={() => onMutate("COMPLETED")}
   683→            >
   684→              <LogOut className="size-4" />
   685→              Check out
   686→            </Button>
   687→          )}
   688→          {(s === "COMPLETED" ||
   689→            s === "CANCELLED" ||
   690→            s === "REJECTED" ||
   691→            s === "NO_SHOW") && (
   692→            <Button variant="outline" className="ml-auto" onClick={onClose}>
   693→              Close
   694→            </Button>
   695→          )}
   696→        </DialogFooter>
   697→      </DialogContent>
   698→    </Dialog>
   699→  );
   700→}

   751→function CreateReservationDialog({
   752→  open,
   753→  onClose,
   754→  onCreated,
   755→}: {
   756→  open: boolean;
   757→  onClose: () => void;
   758→  onCreated: () => void;
   759→}) {
   760→  const qc = useQueryClient();
   761→  const { data: roomsData } = useQuery({
   762→    queryKey: ["admin-rooms", "active"],
   763→    queryFn: () => apiFetch<{ rooms: Room[] }>("/api/rooms"),
   764→    enabled: open,
   765→  });
   766→  const rooms = (roomsData?.rooms ?? []).filter((r) => r.isActive);
   767→
   768→  const {
   769→    register,
   770→    handleSubmit,
   771→    watch,
   772→    setValue,
   773→    reset,
   774→    formState: { errors },
   775→  } = useForm<ReservationCreateInput>({
   776→    resolver: zodResolver(reservationCreateSchema),
   777→    defaultValues: {
   778→      roomId: "",
   779→      checkIn: "",
   780→      checkOut: "",
   781→      adults: 2,
   782→      children: 0,
   783→      guest: {
   784→        firstName: "",
   785→        lastName: "",
   786→        email: "",
   787→        phone: "",
   788→        address: "",
   789→        city: "",
   790→        country: "",
   791→      },
   792→      specialRequests: "",
   793→    },
   794→  });
   795→
   796→  const roomId = watch("roomId");
   797→  const checkIn = watch("checkIn");
   798→  const checkOut = watch("checkOut");
   799→  const selectedRoom = rooms.find((r) => r.id === roomId);
   800→  const nights =
   801→    checkIn && checkOut ? nightsBetween(checkIn, checkOut) : 0;
   802→  const total = selectedRoom ? selectedRoom.pricePerNight * nights : 0;
   803→
   804→  const mutation = useMutation({
   805→    mutationFn: (values: ReservationCreateInput) =>
   806→      apiFetch<{ reservation: Reservation }>("/api/reservations", {
   807→        method: "POST",
   808→        body: JSON.stringify({ ...values, source: "WALK_IN" }),
   809→      }),
   810→    onSuccess: (data) => {
   811→      toast.success(
   812→        `Reservation ${data.reservation.referenceNo} created.`
   813→      );
   814→      qc.invalidateQueries({ queryKey: ["admin-reservations"] });
   815→      qc.invalidateQueries({ queryKey: ["admin-dashboard"] });
   816→      reset();
   817→      onCreated();
   818→    },
   819→    onError: (err) => {
   820→      const message =
   821→        err instanceof ApiError
   822→          ? err.message
   823→          : "Couldn't create reservation.";
   824→      toast.error(message);
   825→    },
   826→  });
   827→
   828→  return (
   829→    <Dialog
   830→      open={open}
   831→      onOpenChange={(o) => {
   832→        if (!o) {
   833→          reset();
   834→          onClose();
   835→        }
   836→      }}
   837→    >
   838→      <DialogContent className="max-h-[90vh] max-w-2xl gap-0 overflow-hidden p-0">
   839→        <DialogHeader className="border-b border-border px-6 py-5">
   840→          <DialogTitle className="font-display text-xl font-medium tracking-tight">
   841→            New reservation
   842→          </DialogTitle>
   843→          <DialogDescription>
   844→            Create a walk-in or phone booking. Status starts as Pending.
   845→          </DialogDescription>
   846→        </DialogHeader>
   847→
   848→        <form
   849→          onSubmit={handleSubmit((v) => mutation.mutate(v))}
   850→          className="flex max-h-[70vh] flex-col"
   851→        >
   852→          <div className="flex-1 overflow-y-auto px-6 py-5">
   853→            {/* Guest */}
   854→            <div className="mb-5">
   855→              <h4 className="mb-3 text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   856→                Guest information
   857→              </h4>
   858→              <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   859→                <Field label="First name" error={errors.guest?.firstName?.message}>
   860→                  <Input {...register("guest.firstName")} className="h-10" />
   861→                </Field>
   862→                <Field label="Last name" error={errors.guest?.lastName?.message}>
   863→                  <Input {...register("guest.lastName")} className="h-10" />
   864→                </Field>
   865→                <Field label="Email" error={errors.guest?.email?.message}>
   866→                  <Input
   867→                    type="email"
   868→                    {...register("guest.email")}
   869→                    className="h-10"
   870→                  />
   871→                </Field>
   872→                <Field label="Phone" error={errors.guest?.phone?.message}>
   873→                  <Input {...register("guest.phone")} className="h-10" />
   874→                </Field>
   875→                <Field label="City" error={errors.guest?.city?.message}>
   876→                  <Input {...register("guest.city")} className="h-10" />
   877→                </Field>
   878→                <Field label="Country" error={errors.guest?.country?.message}>
   879→                  <Input {...register("guest.country")} className="h-10" />
   880→                </Field>
   881→              </div>
   882→            </div>
   883→
   884→            {/* Stay */}
   885→            <div className="mb-5">
   886→              <h4 className="mb-3 text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   887→                Stay details
   888→              </h4>
   889→              <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   890→                <Field label="Room" error={errors.roomId?.message}>
   891→                  <Select
   892→                    value={roomId}
   893→                    onValueChange={(v) => setValue("roomId", v)}
   894→                  >
   895→                    <SelectTrigger className="h-10">
   896→                      <SelectValue placeholder="Select a room" />
   897→                    </SelectTrigger>
   898→                    <SelectContent>
   899→                      {rooms.map((r) => (
   900→                        <SelectItem key={r.id} value={r.id}>
   901→                          {r.name} · {formatCurrency(r.pricePerNight)}/night
   902→                        </SelectItem>
   903→                      ))}
   904→                    </SelectContent>
   905→                  </Select>
   906→                </Field>
   907→                <div className="grid grid-cols-2 gap-3">
   908→                  <Field label="Adults" error={errors.adults?.message}>
   909→                    <Input
   910→                      type="number"
   911→                      min={1}
   912→                      {...register("adults", { valueAsNumber: true })}
   913→                      className="h-10"
   914→                    />
   915→                  </Field>
   916→                  <Field label="Children" error={errors.children?.message}>
   917→                    <Input
   918→                      type="number"
   919→                      min={0}
   920→                      {...register("children", { valueAsNumber: true })}
   921→                      className="h-10"
   922→                    />
   923→                  </Field>
   924→                </div>
   925→                <Field label="Check in" error={errors.checkIn?.message}>
   926→                  <Input
   927→                    type="date"
   928→                    {...register("checkIn")}
   929→                    className="h-10"
   930→                  />
   931→                </Field>
   932→                <Field label="Check out" error={errors.checkOut?.message}>
   933→                  <Input
   934→                    type="date"
   935→                    {...register("checkOut")}
   936→                    className="h-10"
   937→                  />
   938→                </Field>
   939→              </div>
   940→            </div>
   941→
   942→            {/* Special requests */}
   943→            <Field
   944→              label="Special requests (optional)"
   945→              error={errors.specialRequests?.message}
   946→            >
   947→              <Textarea
   948→                {...register("specialRequests")}
   949→                rows={2}
   950→                placeholder="Late check-in, birthday cake, extra towels…"
   951→              />
   952→            </Field>
   953→
   954→            {/* Price summary */}
   955→            {selectedRoom && nights > 0 && (
   956→              <div className="mt-5 rounded-lg border border-border bg-sand/40 p-4">
   957→                <div className="flex items-center justify-between text-sm">
   958→                  <span className="text-muted-foreground">
   959→                    {formatCurrency(selectedRoom.pricePerNight)} × {nights}{" "}
   960→                    {nights === 1 ? "night" : "nights"}
   961→                  </span>
   962→                  <span className="font-medium text-foreground">
   963→                    {formatCurrency(total)}
   964→                  </span>
   965→                </div>
   966→                <div className="mt-1.5 flex items-center justify-between border-t border-border pt-1.5 text-base font-medium">
   967→                  <span>Total</span>
   968→                  <span>{formatCurrency(total)}</span>
   969→                </div>
   970→              </div>
   971→            )}
   972→          </div>
   973→
   974→          <DialogFooter className="border-t border-border px-6 py-4">
   975→            <Button
   976→              type="button"
   977→              variant="outline"
   978→              onClick={() => {

   980→                onClose();

   985→            <Button
   986→              type="submit"
   987→              className="bg-primary text-white hover:bg-primary/90"
   988→              disabled={mutation.isPending}
   989→            >
   990→              {mutation.isPending ? "Creating…" : "Create reservation"}
   991→            </Button>
   992→          </DialogFooter>
   993→        </form>
   994→      </DialogContent>
   995→    </Dialog>
   996→  );
   997→}
