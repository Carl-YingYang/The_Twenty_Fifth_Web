   385→          setCreating(false);
   386→          qc.invalidateQueries({ queryKey: ["admin-reservations"] });
   387→          qc.invalidateQueries({ queryKey: ["admin-dashboard"] });
   388→        }}
   389→      />
   390→    </AdminLayout>
   391→  );

   393→
   394→function RowActions({
   395→  reservation,
   396→  onView,
   397→  onMutate,
   398→  pending,
   399→  fullWidth,
   400→}: {
   401→  reservation: Reservation;
   402→  onView: () => void;
   403→  onMutate: (status: BookingStatus) => void;
   404→  pending: boolean;
   405→  fullWidth?: boolean;
   406→}) {
   407→  const s = reservation.status as BookingStatus;
   408→  return (
   409→    <div
   410→      className={cn(
   411→        "flex items-center justify-end gap-1.5",
   412→        fullWidth && "w-full"
   413→      )}
   414→    >
   415→      {s === "PENDING" && (
   416→        <>
   417→          <Button
   418→            size="sm"
   419→            className="h-8 bg-emerald-600 text-white hover:bg-emerald-700"
   420→            disabled={pending}
   421→            onClick={() => onMutate("CONFIRMED")}
   422→          >
   423→            <Check className="size-3.5" />
   424→            Approve
   425→          </Button>
   426→          <Button
   427→            size="sm"
   428→            variant="outline"
   429→            className="h-8 border-red-300 text-red-700 hover:bg-red-50"
   430→            disabled={pending}
   431→            onClick={() => onMutate("REJECTED")}
   432→          >
   433→            <X className="size-3.5" />
   434→            Decline
   435→          </Button>
   436→        </>
   437→      )}
   438→      {s === "CONFIRMED" && (
   439→        <Button
   440→          size="sm"
   441→          className="h-8 bg-primary text-white hover:bg-primary/90"
   442→          disabled={pending}
   443→          onClick={() => onMutate("CHECKED_IN")}
   444→        >
   445→          <LogIn className="size-3.5" />
   446→          Check in
   447→        </Button>
   448→      )}
   449→      {s === "CHECKED_IN" && (
   450→        <Button
   451→          size="sm"
   452→          className="h-8 bg-primary text-white hover:bg-primary/90"
   453→          disabled={pending}
   454→          onClick={() => onMutate("COMPLETED")}
   455→        >
   456→          <LogOut className="size-3.5" />
   457→          Check out
   458→        </Button>
   459→      )}
   460→      {(s === "COMPLETED" ||
   461→        s === "CANCELLED" ||
   462→        s === "REJECTED" ||
   463→        s === "NO_SHOW") && (
   464→        <Button
   465→          size="sm"
   466→          variant="ghost"
   467→          className="h-8 text-foreground hover:bg-muted"
   468→          onClick={onView}
   469→        >
   470→          <Eye className="size-3.5" />
   471→          View
   472→        </Button>
   473→      )}
   474→      <Button
   475→        size="sm"
   476→        variant="ghost"
   477→        className={cn("h-8 text-foreground hover:bg-muted", fullWidth && "flex-1")}
   478→        onClick={onView}
   479→      >
   480→        <Eye className="size-3.5" />
   481→        Details
   482→      </Button>
   483→    </div>
   484→  );
   485→}

   597→          {reservation.specialRequests && (
   598→            <Section title="Special requests">
   599→              <p className="rounded-lg bg-muted/50 p-3 text-sm text-foreground">
   600→                {reservation.specialRequests}
   601→              </p>
   602→            </Section>
   603→          )}
   604→
   605→          {/* Status timeline */}
   606→          <Section title="Timeline">
   607→            <ul className="space-y-2 text-xs">
   608→              <TimelineItem
   609→                label="Booking received"
   610→                date={reservation.createdAt}
   611→                done
   612→              />
   613→              <TimelineItem
   614→                label="Confirmed"
   615→                date={reservation.confirmedAt}
   616→                done={!!reservation.confirmedAt}
   617→              />
   618→              <TimelineItem
   619→                label="Checked in"
   620→                date={reservation.checkedInAt}
   621→                done={!!reservation.checkedInAt}
   622→              />
   623→              <TimelineItem
   624→                label="Checked out"
   625→                date={reservation.checkedOutAt}
   626→                done={!!reservation.checkedOutAt}
   627→              />
   628→            </ul>
   629→          </Section>
   630→        </div>
   631→
   632→        <DialogFooter className="flex-row flex-wrap gap-2 border-t border-border px-6 py-4">
   633→          {s === "PENDING" && (
   634→            <>
   635→              <Button
   636→                variant="outline"
   637→                className="border-red-300 text-red-700 hover:bg-red-50"
   638→                disabled={pending}
   639→                onClick={() => onMutate("REJECTED")}
   640→              >
   641→                <UserX className="size-4" />
   642→                Decline
   643→              </Button>
   644→              <Button
   645→                className="ml-auto bg-emerald-600 text-white hover:bg-emerald-700"
   646→                disabled={pending}
   647→                onClick={() => onMutate("CONFIRMED")}
   648→              >
   649→                <Check className="size-4" />
   650→                Approve
   651→              </Button>
   652→            </>
   653→          )}
   654→          {s === "CONFIRMED" && (
   655→            <Button
   656→              className="ml-auto bg-primary text-white hover:bg-primary/90"
   657→              disabled={pending}
   658→              onClick={() => onMutate("CHECKED_IN")}
   659→            >
   660→              <LogIn className="size-4" />
   661→              Check in
   662→            </Button>
   663→          )}
   664→          {s === "CHECKED_IN" && (
   665→            <Button
   666→              className="ml-auto bg-primary text-white hover:bg-primary/90"
   667→              disabled={pending}
   668→              onClick={() => onMutate("COMPLETED")}
   669→            >
   670→              <LogOut className="size-4" />
   671→              Check out
   672→            </Button>
   673→          )}
   674→          {(s === "COMPLETED" ||
   675→            s === "CANCELLED" ||
   676→            s === "REJECTED" ||
   677→            s === "NO_SHOW") && (
   678→            <Button variant="outline" className="ml-auto" onClick={onClose}>
   679→              Close
   680→            </Button>
   681→          )}
   682→        </DialogFooter>
   683→      </DialogContent>
   684→    </Dialog>
   685→  );
   686→}

   737→function CreateReservationDialog({
   738→  open,
   739→  onClose,
   740→  onCreated,
   741→}: {
   742→  open: boolean;
   743→  onClose: () => void;
   744→  onCreated: () => void;
   745→}) {
   746→  const qc = useQueryClient();
   747→  const { data: roomsData } = useQuery({
   748→    queryKey: ["admin-rooms", "active"],
   749→    queryFn: () => apiFetch<{ rooms: Room[] }>("/api/rooms"),
   750→    enabled: open,
   751→  });
   752→  const rooms = (roomsData?.rooms ?? []).filter((r) => r.isActive);
   753→
   754→  const {
   755→    register,
   756→    handleSubmit,
   757→    watch,
   758→    setValue,
   759→    reset,
   760→    formState: { errors },
   761→  } = useForm<ReservationCreateInput>({
   762→    resolver: zodResolver(reservationCreateSchema),
   763→    defaultValues: {
   764→      roomId: "",
   765→      checkIn: "",
   766→      checkOut: "",
   767→      adults: 2,
   768→      children: 0,
   769→      guest: {
   770→        firstName: "",
   771→        lastName: "",
   772→        email: "",
   773→        phone: "",
   774→        address: "",
   775→        city: "",
   776→        country: "",
   777→      },
   778→      specialRequests: "",
   779→    },
   780→  });
   781→
   782→  const roomId = watch("roomId");
   783→  const checkIn = watch("checkIn");
   784→  const checkOut = watch("checkOut");
   785→  const selectedRoom = rooms.find((r) => r.id === roomId);
   786→  const nights =
   787→    checkIn && checkOut ? nightsBetween(checkIn, checkOut) : 0;
   788→  const total = selectedRoom ? selectedRoom.pricePerNight * nights : 0;
   789→
   790→  const mutation = useMutation({
   791→    mutationFn: (values: ReservationCreateInput) =>
   792→      apiFetch<{ reservation: Reservation }>("/api/reservations", {
   793→        method: "POST",
   794→        body: JSON.stringify({ ...values, source: "WALK_IN" }),
   795→      }),
   796→    onSuccess: (data) => {
   797→      toast.success(
   798→        `Reservation ${data.reservation.referenceNo} created.`
   799→      );
   800→      qc.invalidateQueries({ queryKey: ["admin-reservations"] });
   801→      qc.invalidateQueries({ queryKey: ["admin-dashboard"] });
   802→      reset();
   803→      onCreated();
   804→    },
   805→    onError: (err) => {
   806→      const message =
   807→        err instanceof ApiError
   808→          ? err.message
   809→          : "Couldn't create reservation.";
   810→      toast.error(message);
   811→    },
   812→  });
   813→
   814→  return (
   815→    <Dialog
   816→      open={open}
   817→      onOpenChange={(o) => {
   818→        if (!o) {
   819→          reset();
   820→          onClose();
   821→        }
   822→      }}
   823→    >
   824→      <DialogContent className="max-h-[90vh] max-w-2xl gap-0 overflow-hidden p-0">
   825→        <DialogHeader className="border-b border-border px-6 py-5">
   826→          <DialogTitle className="font-display text-xl font-medium tracking-tight">
   827→            New reservation
   828→          </DialogTitle>
   829→          <DialogDescription>
   830→            Create a walk-in or phone booking. Status starts as Pending.
   831→          </DialogDescription>
   832→        </DialogHeader>
   833→
   834→        <form
   835→          onSubmit={handleSubmit((v) => mutation.mutate(v))}
   836→          className="flex max-h-[70vh] flex-col"
   837→        >
   838→          <div className="flex-1 overflow-y-auto px-6 py-5">
   839→            {/* Guest */}
   840→            <div className="mb-5">
   841→              <h4 className="mb-3 text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   842→                Guest information
   843→              </h4>
   844→              <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   845→                <Field label="First name" error={errors.guest?.firstName?.message}>
   846→                  <Input {...register("guest.firstName")} className="h-10" />
   847→                </Field>
   848→                <Field label="Last name" error={errors.guest?.lastName?.message}>
   849→                  <Input {...register("guest.lastName")} className="h-10" />
   850→                </Field>
   851→                <Field label="Email" error={errors.guest?.email?.message}>
   852→                  <Input
   853→                    type="email"
   854→                    {...register("guest.email")}
   855→                    className="h-10"
   856→                  />
   857→                </Field>
   858→                <Field label="Phone" error={errors.guest?.phone?.message}>
   859→                  <Input {...register("guest.phone")} className="h-10" />
   860→                </Field>
   861→                <Field label="City" error={errors.guest?.city?.message}>
   862→                  <Input {...register("guest.city")} className="h-10" />
   863→                </Field>
   864→                <Field label="Country" error={errors.guest?.country?.message}>
   865→                  <Input {...register("guest.country")} className="h-10" />
   866→                </Field>
   867→              </div>
   868→            </div>
   869→
   870→            {/* Stay */}
   871→            <div className="mb-5">
   872→              <h4 className="mb-3 text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   873→                Stay details
   874→              </h4>
   875→              <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">

   877→                  <Select
   878→                    value={roomId}
   879→                    onValueChange={(v) => setValue("roomId", v)}
   880→                  >
   881→                    <SelectTrigger className="h-10">

   890→                    </SelectContent>

   892→                </Field>

   895→                    <Input
   896→                      type="number"
   897→                      min={1}
   898→                      {...register("adults", { valueAsNumber: true })}
   899→                      className="h-10"

   904→                      type="number"
   905→                      min={0}
   906→                      {...register("children", { valueAsNumber: true })}
   907→                      className="h-10"
   908→                    />
   909→                  </Field>
   910→                </div>
   911→                <Field label="Check in" error={errors.checkIn?.message}>
   912→                  <Input
   913→                    type="date"
   914→                    {...register("checkIn")}
   915→                    className="h-10"
   916→                  />
   917→                </Field>
   918→                <Field label="Check out" error={errors.checkOut?.message}>
   919→                  <Input
   920→                    type="date"
   921→                    {...register("checkOut")}
   922→                    className="h-10"
   923→                  />
   924→                </Field>
   925→              </div>
   926→            </div>
   927→
   928→            {/* Special requests */}
   929→            <Field
   930→              label="Special requests (optional)"
   931→              error={errors.specialRequests?.message}
   932→            >
   933→              <Textarea
   934→                {...register("specialRequests")}
   935→                rows={2}
   936→                placeholder="Late check-in, birthday cake, extra towels…"
   937→              />
   938→            </Field>
   939→
   940→            {/* Price summary */}
   941→            {selectedRoom && nights > 0 && (
   942→              <div className="mt-5 rounded-lg border border-border bg-sand/40 p-4">
   943→                <div className="flex items-center justify-between text-sm">

   948→                  <span className="font-medium text-foreground">
   949→                    {formatCurrency(total)}
   950→                  </span>

   963→              variant="outline"
